# snouty runs modules
#
# runs modules lists a run's modules (`GET /api/v0/runs/{run_id}/modules`),
# each with the programs that loaded it (`.../modules/{module_id}/instances`).

# A run with instrumented modules lists them, then the counts and the run
# status on stderr.
mock-runs-server
snouty --json runs list --status completed --limit 100
env_from_json_find run_id launcher!=debugging -- runs modules $R_run_id
snouty runs modules $R_run_id
stdout '^MODULE +NAME +PROGRAM +CONTAINER +LANG +LOCATIONS +SYMBOLS$'
stderr '^\d+ modules? in \d+ containers? · run completed$'
snouty --json runs modules $R_run_id
stdout '^\{.*"module_id":.*"num_locations":\d+.*"instances":\[\{.*"container_name":.*"program_path":.*\]\}$'
