# snouty runs usage
#
# Show the usage hours of runs, newest first (`GET /api/v0/usage`). With
# --rollup, show the total of each day, week, or month instead. Times and
# periods use the local timezone, or UTC with --utc.

# Like every runs command, usage requires an API key.
! snouty runs usage
stderr 'No Antithesis credentials found.*'

# --rollup takes day, week, or month.
! snouty runs usage --rollup year
stderr 'possible values: day, week, month'

# --after and --before take a date or an RFC 3339 time.
! snouty runs usage --after 2026-13-01
stderr 'expected a date such as 2026-10-01, or an RFC 3339 time'

# The mock usage entries fall on different days in UTC and in UTC+9, so a
# fixed timezone makes the expected times and periods exact.
env TZ=Asia/Tokyo

# Without --rollup, each run is one row: its run id, the start and end of its
# usage in the local timezone, and its hours to the minute, as `Xh Ym`. The
# command reads every page until the limit.
mock-runs-server
snouty runs usage
stdout '^(RUN ID +STARTED +ENDED +HOURS|No usage found\.)$'
[!staging] stdout '^usage-1 +2026-03-03 01:00:00 +2026-03-03 02:30:00 +10h 30m$'
[!staging] stdout '^usage-3 +2026-03-01 05:00:00 +2026-03-01 05:15:00 +1h 12m$'
[!staging] stdout '^usage-5 +2026-01-15 09:00:00 +2026-01-15 15:00:00 +100h 0m$'
[!staging] stdout -count=6 '^.+$'

# The --json rows have the fields of the usage endpoint, with the times in
# RFC 3339 in the local timezone.
mock-runs-server
snouty --json runs usage -n 1
! stdout '^[^{]'
[!staging] stdout '^\{"run_id":"usage-1","hours":10.5,"start_time":"2026-03-03T01:00:00\+09:00","end_time":"2026-03-03T02:30:00\+09:00"\}$'
[!staging] stdout -count=1 '^.+$'

# --utc shows the times in UTC.
mock-runs-server
[!staging] snouty runs usage --utc -n 1
[!staging] stdout '^usage-1 +2026-03-02 16:00:00 +2026-03-02 17:30:00 +10h 30m$'
mock-runs-server
[!staging] snouty --json runs usage --utc -n 1
[!staging] stdout '"start_time":"2026-03-02T16:00:00Z","end_time":"2026-03-02T17:30:00Z"'

# --limit caps the rows, and a note on stderr says that more can exist.
# Without --after, the rows are the newest runs.
mock-runs-server
[!staging] snouty runs usage -n 2
[!staging] stdout '^usage-2 '
[!staging] ! stdout 'usage-3'
[!staging] stderr 'Showing up to 2 results'

# With --after alone, the rows are the oldest runs that ended at or after it,
# shown newest first. The endpoint sends the newest runs first, so the command
# reads windows of end times forward from --after, the first one a day long,
# until they hold the limit.
mock-runs-server
[!staging] snouty --verbose runs usage --after 2026-02-01 -n 2
[!staging] stdout '(?s)^RUN ID .*\nusage-3 .*\nusage-4 [^\n]*$'
[!staging] ! stdout 'usage-2'
[!staging] stderr 'Showing up to 2 results'
[!staging] stderr '^> GET .*/api/v0/usage\?.*end_time=2026-02-01T15%3A00%3A00Z'

# With both --after and --before, every run between them shows, and --limit
# is not allowed.
mock-runs-server
! snouty runs usage --after 2026-02-01 --before 2026-03-01 -n 2
stderr '--limit cannot be used with both --after and --before'

# --after and --before select on the end time of a run: --after is
# inclusive, and --before is not.
mock-runs-server
[!staging] snouty runs usage --after 2026-02-01 --before 2026-03-01T12:00:00Z
[!staging] stdout '^usage-2 '
[!staging] stdout '^usage-3 '
[!staging] stdout '^usage-4 '
[!staging] ! stdout 'usage-1'
[!staging] ! stdout 'usage-5'
# usage-3 started before 20:10 and ended after it.
mock-runs-server
[!staging] snouty runs usage --after 2026-02-28T20:10:00Z
[!staging] stdout '^usage-3 '
[!staging] ! stdout 'usage-4'

# A date is the start of that day in the local timezone, or in UTC with
# --utc. usage-3 ended at 20:15 UTC on 2026-02-28, which is 2026-03-01 in
# UTC+9.
mock-runs-server
[!staging] snouty runs usage --before 2026-03-01
[!staging] ! stdout 'usage-3'
[!staging] stdout '^usage-5 '
mock-runs-server
[!staging] snouty runs usage --utc --before 2026-03-01
[!staging] stdout '^usage-3 '

# With --rollup, each row is one period: its last day, the number of runs
# that ended in it, and the sum of their hours. Newest period first. The rows
# are the --limit periods that end with the current period, and a period with
# no runs shows 0.
mock-runs-server
snouty runs usage --rollup day
stdout '^DATE +RUNS +HOURS$'
stdout -count=11 '^.+$'
stderr '^Showing up to 10 periods\. Additional periods may be available\.$'

# --before sets the newest period: the one that holds the instant before it.
# usage-4 counts on 2026-02-28, the day that it ended. HOURS is rounded to the
# minute: 2.1 + 1.2 hours shows as 3h 18m.
mock-runs-server
[!staging] snouty runs usage --rollup day -n 5 --before 2026-03-04
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-03 +1 +10h 30m\n2026-03-02 +0 +0h 0m\n2026-03-01 +2 +3h 18m\n2026-02-28 +1 +4h 0m\n2026-02-27 +0 +0h 0m$'
[!staging] stdout -count=6 '^.+$'
[!staging] stderr 'Showing up to 5 periods'

# A week starts on Monday and ends on Sunday.
mock-runs-server
[!staging] snouty runs usage --rollup week -n 3 --before 2026-03-04
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-08 +1 +10h 30m\n2026-03-01 +3 +7h 18m\n2026-02-22 +0 +0h 0m$'
[!staging] stdout -count=4 '^.+$'

# usage-3 ended in February in UTC, but in March in the local timezone. The
# JSON hours of a period are rounded to hundredths.
mock-runs-server
snouty --json runs usage --rollup month -n 3
! stdout '^[^{]'
stdout -count=3 '^.+$'
mock-runs-server
[!staging] snouty --json runs usage --rollup month -n 3 --before 2026-03-04
[!staging] stdout '(?s)^\{"date":"2026-03-31","runs":3,"hours":13.8\}\n\{"date":"2026-02-28","runs":1,"hours":4.0\}\n\{"date":"2026-01-31","runs":1,"hours":100.0\}$'
[!staging] stdout -count=3 '^.+$'

# With --utc, the periods are UTC days, weeks, or months.
mock-runs-server
[!staging] snouty runs usage --rollup month -n 3 --utc --before 2026-03-04
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-31 +2 +12h 36m\n2026-02-28 +2 +5h 12m\n2026-01-31 +1 +100h 0m$'

# With both --after and --before, the rows are every period from the one that
# holds --after to the one before --before, with no limit note. The first
# period counts only the runs inside the filter: usage-4 ended after the start
# of 2026-02-28, and usage-5 did not. The request asks only for the runs of the
# shown periods, so the old history is not read.
mock-runs-server
[!staging] snouty --verbose runs usage --rollup month --after 2026-02-28 --before 2026-03-04
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-31 +3 +13h 48m\n2026-02-28 +1 +4h 0m$'
[!staging] stdout -count=3 '^.+$'
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-27T15%3A00%3A00Z'
[!staging] ! stderr 'Showing up to'
mock-runs-server
[!staging] snouty --verbose runs usage --rollup week -n 2 --before 2026-03-04
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-22T15%3A00%3A00Z'

# With --after alone, the rows are the --limit periods from the one that holds
# --after forward, newest first. One window of end times covers them all.
mock-runs-server
[!staging] snouty --verbose runs usage --rollup day --after 2026-02-27 -n 3
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-01 +2 +3h 18m\n2026-02-28 +1 +4h 0m\n2026-02-27 +0 +0h 0m$'
[!staging] stderr 'Showing up to 3 periods'
[!staging] stderr '^> GET .*/api/v0/usage\?.*end_time=2026-03-01T15%3A00%3A00Z.*start_time=2026-02-26T15%3A00%3A00Z'

# No usage prints a message, or no rows with --json. A rollup shows its
# periods with 0 runs.
[!staging] mock-runs-server empty
[!staging] snouty runs usage
[!staging] stdout '^No usage found.$'
[!staging] mock-runs-server empty
[!staging] snouty runs usage --rollup week -n 2 --before 2026-03-04
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-08 +0 +0h 0m\n2026-03-01 +0 +0h 0m$'
[!staging] mock-runs-server empty
[!staging] snouty --json runs usage
[!staging] ! stdout .
