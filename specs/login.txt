# snouty login — non-interactive paths

# A supplied-but-empty --tenant is rejected by the `validate_non_empty` value
# parser at argument-parsing time — before any prompt, stdin read, or write — so
# the command fails outright with a nonzero exit. (No stdin is provided,
# precisely to prove nothing is read: parsing fails first.)
isolate-home empty_args
! snouty login --tenant ""
stderr '.*Value may not be empty or whitespace.*'

# The same for a supplied-but-empty --registry.
! snouty login --registry ""
stderr '.*Value may not be empty or whitespace.*'

# And for the global --profile flag given an empty value.
! snouty --profile "" login
stderr '.*Value may not be empty or whitespace.*'

# Without a TTY, login saves the tenant and registry, skips credential
# storage, and prints the summary as whole lines, so a script can match a line
# that is longer than a terminal.
isolate-home summary
snouty login --tenant spec-tenant --registry registry.example.com/a-long-registry-path
stdout '^Saved tenant `spec-tenant` and registry `registry.example.com/a-long-registry-path`\.$'
stdout '^Skipped credential storage — snouty will read credentials from the environment, such as ANTITHESIS_API_KEY\.$'
stdout '^Run `snouty doctor` to verify your setup\.$'
file ~/.config/snouty/settings.toml 'registry = "registry.example.com/a-long-registry-path"'

# --repository is an alias for --registry.
isolate-home repository_flag
snouty login --tenant spec-tenant --repository registry.example.com/old-flag
stdout 'registry `registry.example.com/old-flag`\.$'
file ~/.config/snouty/settings.toml 'registry = "registry.example.com/old-flag"'
