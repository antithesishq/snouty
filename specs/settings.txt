# Settings resolution
#
# Snouty resolves each setting with precedence: environment variable, then
# the file that --settings or SNOUTY_SETTINGS_PATH names, then the global
# settings file. `snouty settings show` prints the resolved values. Each layer
# below uses a distinct value, so the resolved value alone proves which layer
# won. Each row is `<setting>  <value>`.

# 1. snouty reads no settings file in the working directory.
snouty settings show
stdout '^tenant +not set$'

# 2. An explicit --settings file is read, and overrides the global file.
mkdir xdg-config-isolation/snouty
cp global.toml xdg-config-isolation/snouty/settings.toml
snouty settings show
stdout 'tenant.*global-tenant'
stdout 'registry.*global/repo'
snouty --settings other.toml settings show
stdout 'tenant.*other-tenant'
stdout 'registry.*global/repo'

# 3. An environment variable overrides every file.
env ANTITHESIS_TENANT=from-env
snouty --settings other.toml settings show
stdout 'tenant.*from-env'

# 4. The container engine override is reported in the table.
env SNOUTY_CONTAINER_ENGINE=podman
snouty settings show
stdout 'container_engine.*podman'

# 5. The update channel is reported in the table; unset resolves to stable.
snouty settings show
stdout 'update_channel.*stable'
env SNOUTY_UPDATE_CHANNEL=unstable
snouty settings show
stdout 'update_channel.*unstable'

# 6. SNOUTY_SETTINGS_PATH names the settings file, and the --settings flag
#    wins over it.
env SNOUTY_SETTINGS_PATH=env-settings.toml
snouty settings show
stdout 'registry.*env-path-repo'
snouty --settings other-registry.toml settings show
stdout 'registry.*other/repo'

# 7. `repository` is an alias for `registry`, and loads as the registry.
#    When a file holds both keys, `registry` wins.
snouty --settings old-key.toml settings show
stdout '^registry +old-key$'
snouty --settings both-keys.toml settings show
stdout '^registry +new-key$'

# 8. ANTITHESIS_REPOSITORY, an alias for ANTITHESIS_REGISTRY, loads,
#    and wins over the files. When both are set, ANTITHESIS_REGISTRY wins.
env ANTITHESIS_REPOSITORY=old-env
snouty --settings both-keys.toml settings show
stdout '^registry +old-env$'
env ANTITHESIS_REGISTRY=new-env
snouty --settings both-keys.toml settings show
stdout '^registry +new-env$'

# 9. The --profile flag no longer exists.
! snouty --profile prod settings show
stderr 'unexpected argument .--profile.'

# 10. ANTITHESIS_PROFILE fails every command that reads the settings.
env ANTITHESIS_PROFILE=prod
! snouty settings show
stderr 'ANTITHESIS_PROFILE is set'
stderr -literal 'Suggestion: unset ANTITHESIS_PROFILE'
env ANTITHESIS_PROFILE=

-- .snouty.toml --
tenant = "acme"
-- global.toml --
tenant = "global-tenant"
registry = "global/repo"
-- other.toml --
tenant = "other-tenant"
-- other-registry.toml --
registry = "other/repo"
-- env-settings.toml --
registry = "env-path-repo"
-- old-key.toml --
repository = "old-key"
-- both-keys.toml --
repository = "old-key"
registry = "new-key"
