# Project settings file trust
#
# A project settings file (./.snouty.toml) can set base_url and https_proxy,
# which decide where snouty sends the user's credentials. A cloned repository
# can ship one. So snouty uses a project file only after the user trusts it
# with `snouty config trust`, and asks again when its content changes.
# `snouty config deny` makes snouty ignore the file. A file named with
# --settings or SNOUTY_SETTINGS_PATH is trusted, because the user chose it.
#
# Pin the config dir, so that the trust record stays the same after a `cd`.
set-env XDG_CONFIG_HOME ${WORK}/config

# 1. An untrusted project file stops every command that reads settings. The
#    error names the file and each way to continue.
! snouty config show
stderr 'is not trusted'
stderr '\.snouty\.toml'
stderr 'snouty config trust'
stderr 'snouty config deny'
stderr 'SNOUTY_CONFIG_TRUST=all'
! stdout .
! snouty doctor --offline
stderr 'is not trusted'

# 2. SNOUTY_CONFIG_TRUST=all skips the check, and records nothing.
env SNOUTY_CONFIG_TRUST=all
snouty config show
stdout 'tenant +acme'
env SNOUTY_CONFIG_TRUST=
! snouty config show
stderr 'is not trusted'

# 3. Any other SNOUTY_CONFIG_TRUST value is an error.
env SNOUTY_CONFIG_TRUST=yes
! snouty config show
stderr 'SNOUTY_CONFIG_TRUST'
stderr '`all`'
env SNOUTY_CONFIG_TRUST=

# 4. `config trust` trusts ./.snouty.toml, and snouty then uses it.
snouty config trust
stderr 'Trusted .*\.snouty\.toml'
snouty config show
stdout 'tenant +acme'
stdout 'base_url +https://acme\.antithesis\.com'

# 5. A trusted file whose content changes is not trusted any more.
cp evil.toml .snouty.toml
! snouty config show
stderr 'changed since you trusted it'
stderr 'snouty config trust'
snouty config trust
snouty config show
stdout 'base_url +https://evil\.example\.com'

# 6. `config deny` makes snouty ignore the file, also after it changes.
snouty config deny
stderr 'Denied .*\.snouty\.toml'
snouty config show
stdout 'tenant +not set'
cp acme.toml .snouty.toml
snouty config show
stdout 'tenant +not set'

# 7. `config trust` after `config deny` uses the file again.
snouty config trust
snouty config show
stdout 'tenant +acme'

# 8. `config trust` and `config deny` take an explicit path.
cd sub
! snouty config show
stderr 'is not trusted'
cd .
snouty config trust sub/.snouty.toml
cd sub
snouty config show
stdout 'tenant +sub-tenant'
cd .
snouty config deny sub/.snouty.toml
cd sub
snouty config show
stdout 'tenant +not set'
cd .

# 9. A file named with --settings or SNOUTY_SETTINGS_PATH needs no trust.
snouty --settings chosen.toml config show
stdout 'tenant +chosen-tenant'
env SNOUTY_SETTINGS_PATH=chosen.toml
snouty config show
stdout 'tenant +chosen-tenant'
env SNOUTY_SETTINGS_PATH=

# 10. `config trust` fails for a file that does not exist.
! snouty config trust missing.toml
stderr 'missing\.toml'
stderr 'not found'

# 11. `config deny` refuses a path that is not a file.
! snouty config deny sub
stderr 'is not a settings file'

# 12. `config trust` and `config deny` refuse --settings, which names a file
#     that is trusted already.
! snouty --settings chosen.toml config trust
stderr 'ignore --settings'
! snouty --settings chosen.toml config deny
stderr 'ignore --settings'

# 13. A bad SNOUTY_CONFIG_TRUST value fails also where no project file exists.
cd empty
env SNOUTY_CONFIG_TRUST=yes
! snouty config show
stderr 'SNOUTY_CONFIG_TRUST'
env SNOUTY_CONFIG_TRUST=
cd .

# 14. An invalid trust file is an error that tells how to recover.
cp broken-trust.toml config/snouty/trust.toml
! snouty config show
stderr 'trust file .* is not valid'
stderr 'Delete the file'

-- .snouty.toml --
tenant = "acme"
-- acme.toml --
tenant = "acme"
-- evil.toml --
tenant = "acme"
base_url = "https://evil.example.com"
-- chosen.toml --
tenant = "chosen-tenant"
-- empty/.keep --
-- broken-trust.toml --
[files."/x"]
decision = "maybe"
-- sub/.snouty.toml --
tenant = "sub-tenant"
