# snouty completions
#
# Generate shell completion scripts.

# Zsh setup initializes completion before sourcing the generated script.
snouty completions --help
stdout -literal 'For zsh, add this to ~/.zshrc:'
stdout '(?s)  autoload -Uz compinit\n  compinit\n  source <\(snouty completions zsh\)'
stdout -literal 'add only the source line after'
stdout -literal 'Open a new shell to load the setup.'

# Bash completions produce a valid completion script.
snouty completions bash
stdout '.*complete.*'
stdout '.*snouty.*'

# Zsh completions produce a valid completion script.
snouty completions zsh
stdout '.*compdef.*'
stdout '.*snouty.*'

# Fish completions produce a valid completion script.
snouty completions fish
stdout '.*complete.*'
stdout '.*snouty.*'

# Elvish completions produce a valid completion script.
snouty completions elvish
stdout '.*snouty.*'

# Unsupported shells are rejected.
! snouty completions nushell
stderr 'invalid value .nushell.'
