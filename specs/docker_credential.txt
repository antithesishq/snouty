# snouty docker-credential: the Docker credential helper protocol for the
# tenant registry, registry.<tenant>.antithesis.com.
#
# Docker writes the server URL on stdin and reads JSON from stdout. The
# username is always snouty-token, and the secret is the Antithesis
# credential of the selected profile. Settings files resolve as for every
# command: --settings or SNOUTY_SETTINGS_PATH takes the place of
# ./.snouty.toml, and the global settings file applies.
#
# A host that is not registry.<tenant>.antithesis.com is not found, and the
# helper reads no settings for it.
#
# 1. With --profile or ANTITHESIS_PROFILE, the helper uses only that
#    profile, or fails when it does not resolve. When its tenant registry is
#    not the requested host, the credential is not found.
# 2. Otherwise, the first profile whose tenant registry is the requested
#    host and that has a credential: the top-level defaults, then the
#    profiles of the project settings file, then the other profiles of the
#    global settings file, each file in name order. A profile that fails to
#    resolve is skipped. When profiles match but none has a credential, the
#    helper fails and names them.

# The settings and the credentials in this file are local fixtures.
[staging] skip

env RUST_LOG=

# One match: the top-level defaults of the global settings file
# use registry.acme.antithesis.com, and the defaults have their own stored
# API key. The profile `broken` fails to resolve, and is skipped.
isolate-home stored
substitute ~/.config/snouty/credentials.toml
stdin acme.txt
snouty docker-credential get
stdout -count=1 '^\{"ServerURL":"registry.acme.antithesis.com","Username":"snouty-token","Secret":"${FAKE_API_KEY_A@R}"\}$'
! stderr .

# A scheme, a path and capital letters do not change the host. ServerURL
# echoes the request as Docker gave it.
stdin beta_url.txt
snouty docker-credential get
stdout '^\{"ServerURL":"https://Registry.beta.antithesis.com/v2/","Username":"snouty-token","Secret":"${FAKE_API_KEY_B@R}"\}$'

# A profile whose repository is on another registry does not use its
# tenant registry: the exact "not found" message, exit status 1, and no
# noise on stderr.
stdin gamma.txt
! snouty docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stdout '\{'
! stderr .

# A host that no profile uses is not found either.
stdin elsewhere.txt
! snouty docker-credential get
stdout '^credentials not found in native keychain$'

# Several matches: the first one wins, with no error. The
# defaults come before the profile acme-b, which has the key B.
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'
! stderr .

# Named profiles come in name order: one, which uses the key of the
# defaults, comes before two, which has the key B.
stdin zeta.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'
! stderr .

# The helper reads the settings of the directory where Docker runs it, as
# every command does. There, ./.snouty.toml replaces the top-level tenant and
# repository of the global settings file, and adds profiles. The user
# trusts the file first, as for every command.
cd project
snouty settings trust
stdin omega.txt
snouty docker-credential get
stdout -count=1 '^\{"ServerURL":"registry.omega.antithesis.com","Username":"snouty-token","Secret":"${FAKE_API_KEY_A@R}"\}$'
! stderr .

stdin sigma.txt
snouty docker-credential get
stdout '"ServerURL":"registry.sigma.antithesis.com".*"Secret":"${FAKE_API_KEY_A@R}"'

# The global defaults no longer use registry.acme.antithesis.com there, so
# the next match, acme-b, wins.
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'

# The profiles of ./.snouty.toml come before the profiles of the global
# settings file: zz, with the key B, comes before one.
stdin zeta.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'

# In another directory, the global settings apply again.
cd .
stdin omega.txt
! snouty docker-credential get
stdout '^credentials not found in native keychain$'

# ANTITHESIS_PROFILE or --profile selects one profile.
env ANTITHESIS_PROFILE=beta

snouty docker-credential get

stdin beta_url.txt

# When the selected profile does not use the host, the credential is not
# found, though the defaults use it.

! snouty docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stderr .
env ANTITHESIS_PROFILE=

stdin acme.txt
! snouty --profile beta docker-credential get
stdout '^credentials not found in native keychain$'

# A selected profile wins over the defaults, which come first in the
# search.

snouty --profile acme-b docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'

# A selected profile that fails to resolve fails the helper. The error goes
# to stdout, where Docker shows it, and to stderr.

! snouty docker-credential get
stdout '^invalid tenant `not a host`'

! stdout '"Secret"'

# A host that is not a tenant registry is not found before the helper reads
# the settings, so the profile that fails to resolve is no error.

! snouty docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stderr .
env ANTITHESIS_PROFILE=

stdin acme.txt
! snouty --profile broken docker-credential get
stdout '^invalid tenant `not a host`'

# store and list fail with one line on stdout, where the engine shows it.
# Docker and Podman call store for a login, and snouty owns the credential.
# Neither engine calls list for a "credHelpers" entry.
substitute store.json
stdin store.json
! snouty docker-credential store
stdout -count=1 '^snouty provides credentials for this registry; to sign in, run `snouty login` or set ANTITHESIS_API_KEY$'
! snouty docker-credential list
stdout -count=1 '^the snouty credential helper is not a general-purpose credential store; use it only as a "credHelpers" entry for registry.<tenant>.antithesis.com, not as "credsStore"$'

# erase reads stdin, erases nothing, and succeeds, because a failed erase
# breaks `podman logout --all` for every registry.
stdin acme.txt
snouty docker-credential erase
! stdout .

# They change nothing: get still serves the stored credential.
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'

# OAuth: get refreshes a token whose expiry it cannot read, and stores the
# new token. A token that expires in more than 2 hours is used with no
# refresh. The tests in src/registry.rs cover the expiry margin.
mock-runs-server
env ANTITHESIS_API_KEY=
env ANTITHESIS_REPOSITORY=registry.testtenant.antithesis.com/app
isolate-home oauth
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_B@R}"'
file ~/.config/snouty/credentials.toml 'antithesis_token = "${FAKE_ACCESS_TOKEN_B@R}"'
file ~/.config/snouty/credentials.toml 'refresh_token = "${FAKE_REFRESH_TOKEN_A@R}"'

# When the refresh fails, the stored token serves while it has not expired,
# with a warning that does not show the token.
env ANTITHESIS_BASE_URL=http://127.0.0.1:9
isolate-home oauth_refresh_fails
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_A@R}"'
stderr '^warning: the OAuth token refresh failed, so the stored token is used until it expires: '
! stderr '${FAKE_ACCESS_TOKEN_A@R}'

# When the stored token has expired too, the refresh error is the result.
isolate-home oauth_expired_refresh

stdin testtenant.txt
! snouty docker-credential get
stdout '^error sending request for url \(http://127.0.0.1:9/auth/cli/refresh\)$'
! stdout '"Secret"'
! stderr 'warning:'
env ANTITHESIS_BASE_URL=
