# snouty runs usage
#
# Show the compute hours of runs (`GET /api/v0/usage`), or of the runs that
# the command names (`GET /api/v0/usage/{run_id}`). With --rollup, show the
# total of each day, week, or month instead. Times and periods use the local
# timezone.

# Like every runs command, usage requires an API key.
! snouty runs usage
stderr 'No Antithesis credentials found.*'

# --rollup takes day, week, or month.
! snouty runs usage --rollup year
stderr 'possible values: day, week, month'

# --after and --before refuse a time that is in none of their forms.
! snouty runs usage --after 2026-13-01
stderr 'expected a date \(2026-10-01\), a time as snouty prints it'

# The mock usage entries fall on different days in UTC and in UTC+9, so a
# fixed timezone makes the expected times and periods exact.
env TZ=JST-9

# Without a filter, the rows are the runs that ended in the last week.
mock-runs-server
snouty --verbose runs usage
stdout '^(RUN ID +STARTED +ENDED +COMPUTE HOURS|No usage found\.)$'
stderr '^> GET .*/api/v0/usage\?.*start_time='
! stderr 'end_time='

# Each run is one row: its run id, the start and end of its usage in the
# local timezone, and its compute hours. The rows are in the order in which
# the runs ended, oldest first, and the command reads every page.
mock-runs-server
[!staging] snouty runs usage --after 2026-01-01
[!staging] stdout '(?s)^RUN ID .*\nusage-5 .*\nusage-4 .*\nusage-3 .*\nusage-2 .*\nusage-1 [^\n]*$'
[!staging] stdout '^usage-1 +2026-03-03 01:00:00 +2026-03-03 02:30:00 +10\.5$'
[!staging] stdout '^usage-3 +2026-03-01 05:00:00 +2026-03-01 05:15:00 +1\.2$'
[!staging] stdout '^usage-5 +2026-01-15 09:00:00 +2026-01-15 15:00:00 +100\.0$'
[!staging] stdout -count=6 '^.+$'

# The JSON times are RFC 3339 in UTC, while the text shows them in the local
# timezone.
mock-runs-server
snouty --json runs usage
! stdout '^[^{]'
[!staging] snouty --json runs usage --after 2026-03-02 --before 2026-03-04
[!staging] stdout '^\{"run_id":"usage-1","hours":10.5,"start_time":"2026-03-02T16:00:00Z","end_time":"2026-03-02T17:30:00Z"\}$'
[!staging] stdout -count=1 '^.+$'

# --after and --before take a relative time.
mock-runs-server
snouty runs usage --after '2 hours ago' --before now
stdout '^(RUN ID +STARTED +ENDED +COMPUTE HOURS|No usage found\.)$'
! snouty runs usage --after '2 fortnights ago'
stderr 'a relative time \(now, 3 days ago\)'

# --after defaults to 1 week before --before.
mock-runs-server
[!staging] snouty --verbose runs usage --before 2026-03-04
[!staging] stdout '(?s)^RUN ID .*\nusage-4 .*\nusage-3 .*\nusage-2 .*\nusage-1 [^\n]*$'
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-24T15%3A00%3A00Z'

# --after and --before select on the end time of a run: --after is
# inclusive, and --before is not.
mock-runs-server
[!staging] snouty runs usage --after 2026-02-01 --before 2026-03-01T12:00:00Z
[!staging] stdout '^usage-2 '
[!staging] stdout '^usage-3 '
[!staging] stdout '^usage-4 '
[!staging] ! stdout 'usage-1'
[!staging] ! stdout 'usage-5'
# usage-3 started before 20:10 and ended after it.
mock-runs-server
[!staging] snouty runs usage --after 2026-02-28T20:10:00Z --before 2026-03-04
[!staging] stdout '^usage-3 '
[!staging] ! stdout 'usage-4'

# A date is the start of that day, and a time as snouty prints it is in the
# local timezone. usage-3 ended at 20:15 UTC on 2026-02-28, which is
# 2026-03-01 05:15:00 in UTC+9.
mock-runs-server
[!staging] snouty runs usage --after 2026-01-01 --before 2026-03-01
[!staging] ! stdout 'usage-3'
[!staging] stdout '^usage-5 '
mock-runs-server
[!staging] snouty runs usage --after '2026-03-01 05:15:00' --before 2026-03-02
[!staging] stdout '^usage-3 '
[!staging] ! stdout 'usage-4'

# With --rollup, each row is one period: its first and last days, the number
# of runs that ended in it, and the sum of their hours. Without a filter, the
# rows are the periods from the one that holds the time 1 week ago to the
# current one, or from 4 weeks ago for weeks.
mock-runs-server
snouty runs usage --rollup day
stdout '^START +END +RUNS +COMPUTE HOURS$'
stdout -count=9 '^.+$'
snouty runs usage --rollup week
stdout -count=6 '^.+$'

# The periods are oldest first, and a period with no runs shows 0. --before
# sets the newest period: the one that holds the instant before it. usage-4
# counts on 2026-02-28, the day that it ended. COMPUTE HOURS has one decimal
# place: 2.1 + 1.2 compute hours shows as 3.3.
mock-runs-server
[!staging] snouty runs usage --rollup day --after 2026-02-27 --before 2026-03-04
[!staging] stdout '(?s)^START +END +RUNS +COMPUTE HOURS\n2026-02-27 +2026-02-27 +0 +0\.0\n2026-02-28 +2026-02-28 +1 +4\.0\n2026-03-01 +2026-03-01 +2 +3\.3\n2026-03-02 +2026-03-02 +0 +0\.0\n2026-03-03 +2026-03-03 +1 +10\.5$'

# A week starts on Monday and ends on Sunday. Without --after, the first
# period is whole: the request starts at the start of that period, not 4
# weeks before --before.
mock-runs-server
[!staging] snouty --verbose runs usage --rollup week --before 2026-03-04
[!staging] stdout '(?s)^START +END +RUNS +COMPUTE HOURS\n2026-02-02 +2026-02-08 +0 +0\.0\n2026-02-09 +2026-02-15 +0 +0\.0\n2026-02-16 +2026-02-22 +0 +0\.0\n2026-02-23 +2026-03-01 +3 +7\.3\n2026-03-02 +2026-03-08 +1 +10\.5$'
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-01T15%3A00%3A00Z'

# usage-3 ended in February in UTC, but in March in the local timezone. The
# JSON hours of a period are rounded to hundredths.
mock-runs-server
snouty --json runs usage --rollup month
stdout '^\{"date":"\d{4}-\d\d-\d\d","runs":\d+,"hours":[\d.]+\}$'
! stdout '^[^{]'
mock-runs-server
[!staging] snouty --json runs usage --rollup month --after 2026-01-01 --before 2026-03-04
[!staging] stdout '(?s)^\{"date":"2026-01-31","runs":1,"hours":100.0\}\n\{"date":"2026-02-28","runs":1,"hours":4.0\}\n\{"date":"2026-03-31","runs":3,"hours":13.8\}$'
[!staging] stdout -count=3 '^.+$'

# When --after cuts a period, that period counts only the runs inside the
# filter: usage-4 ended after the start of 2026-02-28. The request asks only
# for the runs of the shown periods.
mock-runs-server
[!staging] snouty --verbose runs usage --rollup month --after 2026-02-28 --before 2026-03-04
[!staging] stdout '(?s)^START +END +RUNS +COMPUTE HOURS\n2026-02-01 +2026-02-28 +1 +4\.0\n2026-03-01 +2026-03-31 +3 +13\.8$'
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-27T15%3A00%3A00Z'
mock-runs-server
[!staging] snouty runs usage --rollup month --after 2026-02-01 --before 2026-03-04
[!staging] stdout '^2026-02-01 +2026-02-28 +1 +4\.0$'

# No usage prints a message, or no rows with --json. A rollup shows its
# periods with 0 runs.
[!staging] mock-runs-server empty
[!staging] snouty runs usage
[!staging] stdout '^No usage found.$'
[!staging] mock-runs-server empty
[!staging] snouty runs usage --rollup week --after 2026-02-23 --before 2026-03-04
[!staging] stdout '(?s)^START +END +RUNS +COMPUTE HOURS\n2026-02-23 +2026-03-01 +0 +0\.0\n2026-03-02 +2026-03-08 +0 +0\.0$'
[!staging] mock-runs-server empty
[!staging] snouty --json runs usage
[!staging] ! stdout .

# A run from the usage list has a row of its own. An unknown run fails the
# command, after the rows of the other runs.
mock-runs-server
[!staging] snouty --json runs usage --after 2026-03-02 --before 2026-03-04
[staging] snouty --json runs usage
env_from_json 0 run_id
snouty runs usage ${R_run_id}
stdout '^RUN ID +STARTED +ENDED +COMPUTE HOURS$'
stdout '^${R_run_id@R} '
! snouty runs usage ${R_run_id} no-such-run
stdout '^${R_run_id@R} '
stderr 'no usage found for no-such-run'

# With run IDs, each run is one row, in the order given. The runs with no
# usage fail the command, which names them and still prints the other rows.
mock-runs-server
[!staging] snouty runs usage usage-3 usage-1
[!staging] stdout '(?s)^RUN ID .*\nusage-3 .*\nusage-1 [^\n]*$'
mock-runs-server
[!staging] ! snouty --json runs usage usage-3 no-such-run usage-1 other-run
[!staging] stdout -count=2 '^\{"run_id":'
[!staging] stderr 'no usage found for no-such-run, other-run'
[!staging] stderr -literal 'Suggestion: check each run ID and the status of its run with `snouty runs show <run_id>`'
mock-runs-server
[!staging] snouty --json runs usage usage-1
[!staging] stdout '^\{"run_id":"usage-1","hours":10.5,'
[!staging] stdout -count=1 '^.+$'

# Run IDs cannot be used with a filter or --rollup.
! snouty runs usage usage-1 --after 2026-02-01
stderr 'cannot be used with'
! snouty runs usage usage-1 --before 2026-03-01
stderr 'cannot be used with'
! snouty runs usage usage-1 --rollup day
stderr 'cannot be used with'
