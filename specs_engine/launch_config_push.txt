# snouty launch --config — image push and digest pinning
#
# When --config is used with a docker-compose config, the local image store is
# the single source of truth: snouty never pulls. Every service image must be
# present locally and is pinned to its local digest — served from a registry
# that verifiably already has that digest (a manifest-only round trip, no
# pull/push), or tagged with the ANTITHESIS_REPOSITORY prefix and pushed. The
# compose file is then canonicalized and rewritten with digest-pinned image
# references and baked into the config image. The config image itself is
# pinned (name@sha256:digest) in the webhook params. snouty no longer sets
# antithesis.images — the pinned compose file is the source of truth.

# Local build images are tagged with the registry prefix and pushed; the
# config image is pinned. antithesis.images is never set.
mock-server 200 '{"runId":"run-123","statusCode":200}'
build-image --platform linux/amd64 myapp:latest images/myapp
build-image --platform linux/amd64 sidecar:v1 images/sidecar
snouty launch --launcher basic_test -c push_config --duration 30m
stderr 'Pushing image:.*myapp'
stderr 'Pushing image:.*sidecar'
stderr '"antithesis.config_image":.*@sha256:'
! stderr '"antithesis.images"'

# Relaunching after a successful push skips re-pushing: the registry round
# trip confirms the pushed digest is already there. docker-only — podman's
# push digest differs from its local digest (recompression), so podman
# re-pushes by design. The positive skip lines below cover both service images,
# so there are no negated assertions here.
[docker] mock-server 200 '{"runId":"run-123","statusCode":200}'
[docker] snouty launch --launcher basic_test -c push_config --duration 30m
[docker] stderr 'Image already in a registry, skipping push:.*myapp'
[docker] stderr 'Image already in a registry, skipping push:.*sidecar'
[docker] stderr '"antithesis.config_image":.*@sha256:'

# A registry-qualified third-party image that is NOT in the local store fails
# fast — snouty never pulls, so the user must pull it themselves first.
mock-server 200 '{"runId":"run-123","statusCode":200}'
! snouty launch --launcher basic_test -c external_config --duration 30m
stderr '.*some images are not available locally.*'
stderr '.*image: docker.io/library/nginx-snouty-absent:latest.*'
stderr '.*pull or build the missing images, then retry.*'
stderr 'add `--allow-remote-images` to launch without local validation and pinning'
! stderr 'Pushing image:'

# Mixed: even though the build image is present, a missing remote image fails
# the launch before anything is pushed.
mock-server 200 '{"runId":"run-123","statusCode":200}'
build-image --platform linux/amd64 myapp:latest images/myapp
! snouty launch --launcher basic_test -c mixed_config --duration 30m
stderr '.*image: docker.io/library/nginx-snouty-absent:latest.*'
! stderr 'Pushing image:'

# With --allow-remote-images, a missing image does not fail the launch, and the
# local image is still pinned. An earlier case pushed `myapp`, so the registry
# can already serve it.
mock-server 200 '{"runId":"run-123","statusCode":200}'
build-image --platform linux/amd64 myapp:latest images/myapp
snouty launch --launcher basic_test -c mixed_config --duration 30m --allow-remote-images
stderr '^warning: these images are not in the local image store, so snouty cannot validate and pin them:$'
stderr '^  docker.io/library/nginx-snouty-absent:latest$'
stderr '(Pushing image|Image already in a registry, skipping push):.*myapp'
! stderr '(Pushing image|skipping push):.*nginx-snouty-absent'
stderr '"antithesis.config_image":.*@sha256:'

# A service with a `build:` stanza whose image is not local was possibly not
# built, so the launch also warns that it does not run the local code. The
# warning names that service even when another service shares the image.
mock-server 200 '{"runId":"run-123","statusCode":200}'
snouty launch --launcher basic_test -c unbuilt_config --duration 30m --allow-remote-images
stderr '^  snouty-unbuilt-xyz:latest$'
stderr '^warning: these services have a `build:` stanza, but their images are not in the local image store. To launch the local code, run `docker compose build`:$'
stderr '^  app$'
! stderr '^  api$'

# A local image referenced WITHOUT a build stanza, but built before launch, is
# pushed and pinned anyway — local availability is enough.
mock-server 200 '{"runId":"run-123","statusCode":200}'
build-image --platform linux/amd64 snouty-prebuilt-xyz:latest images/myapp
snouty launch --launcher basic_test -c prebuilt_config --duration 30m
stderr 'Pushing image:.*snouty-prebuilt-xyz'
stderr '"antithesis.config_image":.*@sha256:'
! stderr '"antithesis.images"'

# A bare local image that is not in the local store fails fast, rather than
# letting Antithesis fail to pull it partway into a run.
mock-server 200 '{"runId":"run-123","statusCode":200}'
! snouty launch --launcher basic_test -c local_no_build_config --duration 30m
stderr '.*image: localapp:latest.*'
! stderr 'Pushing image:'

# Every service needs an explicit image:, because Antithesis runs compose
# under its own project name. The default tag that `docker compose build`
# gives a build-only service names no image there. The launch fails before
# anything is pushed, even when that tag is built, and with
# --allow-remote-images too.
mock-server 200 '{"runId":"run-123","statusCode":200}'
build-image --platform linux/amd64 buildonly_config-app:latest images/myapp
! snouty launch --launcher basic_test -c buildonly_config --duration 30m
stderr 'every compose service needs an explicit `image:` reference'
stderr 'service .app. has no `image:` field'
! stderr 'Pushing image:'
mock-server 200 '{"runId":"run-123","statusCode":200}'
! snouty launch --launcher basic_test -c buildonly_config --duration 30m --allow-remote-images
stderr 'service .app. has no `image:` field'
! stderr 'Pushing image:'

# A non-amd64 image scheduled for push is rejected before any push happens.
mock-server 200 '{"runId":"run-123","statusCode":200}'
build-image --platform linux/arm64 arm-launch:latest images/arm_launch
! snouty launch --launcher basic_test -c arm_push_config --duration 30m
stderr 'x86-64 \(amd64\).*'
stderr 'image '\''.*arm-launch:latest'\'' has architecture '\''arm64'\''.*'
! stderr 'Pushing image:'

# Kubernetes config: builds and pushes the config image, never sets
# antithesis.images (the platform pulls images from the manifests).
mock-server 200 '{"runId":"run-123","statusCode":200}'
snouty launch --launcher basic_k8s_test -c k8s_config --duration 30m
stderr '"antithesis.config_image":.*@sha256:'
! stderr '"antithesis.images"'

# --allow-remote-images has no effect on a Kubernetes config.
mock-server 200 '{"runId":"run-123","statusCode":200}'
snouty launch --launcher basic_k8s_test -c k8s_config --duration 30m --allow-remote-images
stderr '"antithesis.config_image":.*@sha256:'
! stderr 'not in the local image store'

# With --config, the `vcs.*` params come from the git checkout that holds the
# config dir, not from the current directory.
env GIT_CONFIG_GLOBAL=/dev/null
env GIT_CONFIG_NOSYSTEM=1
exec git -C git_k8s_config init -q -b main
exec git -C git_k8s_config -c user.name=snouty -c user.email=snouty@example.com commit -q --allow-empty -m init
exec git -C git_k8s_config remote add origin https://github.com/acme/widget.git
mock-server 200 '{"runId":"run-123","statusCode":200}'
snouty launch --launcher basic_k8s_test -c git_k8s_config --duration 30m
stderr '"vcs.repo_owner": "acme"'
stderr '"vcs.repo_branch": "main"'
stderr '"vcs.version_id": "[0-9a-f]{40}"'
