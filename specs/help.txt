# snouty version

# version subcommand prints version string.
snouty version
stdout 'snouty \d+\.\d+\.\d+'

# --json prints the version and the commit as one JSON object, with no
# warning that --json has no effect.
snouty --json version
stdout '"version": "\d+\.\d+\.\d+'
stdout '"commit": '
! stderr .

# --version flag prints version string.
snouty --version
stdout 'snouty \d+\.\d+\.\d+'
