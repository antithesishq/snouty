# snouty login — non-interactive paths

# A supplied-but-empty --tenant is rejected by the `validate_non_empty` value
# parser at argument-parsing time — before any prompt, stdin read, or write — so
# the command fails outright with a nonzero exit. (No stdin is provided,
# precisely to prove nothing is read: parsing fails first.)
isolate-home empty_args
! snouty login --tenant ""
stderr '.*Value may not be empty or whitespace.*'

# The same for a supplied-but-empty --repository.
! snouty login --repository ""
stderr '.*Value may not be empty or whitespace.*'

# And for the global --profile flag given an empty value.
! snouty --profile "" login
stderr '.*Value may not be empty or whitespace.*'

# Without a TTY, login signs in with the credential from the environment,
# saves the tenant and the tenant's registry, skips credential storage, and
# prints the summary as whole lines, so a script can match a line that is
# longer than a terminal.
isolate-home summary
mock-runs-server
snouty login --tenant ${ANTITHESIS_TENANT}
stdout '^Saved tenant `${ANTITHESIS_TENANT@R}` and repository `[^`/]+\.[^`]+`\.$'
[!staging] stdout '^Saved tenant `testtenant` and repository `registry.testtenant.antithesis.com`\.$'
stdout '^Skipped credential storage — snouty will read credentials from the environment, such as ANTITHESIS_API_KEY\.$'
stdout '^Run `snouty doctor` to verify your setup\.$'
file ~/.config/snouty/settings.toml 'repository = "[^"]+"'
[!staging] file ~/.config/snouty/settings.toml 'repository = "registry.testtenant.antithesis.com"'

# --repository takes the place of the tenant's registry.
isolate-home repository_flag
snouty login --tenant ${ANTITHESIS_TENANT} --repository registry.example.com/a-long-repository-name
stdout 'repository `registry.example.com/a-long-repository-name`\.$'

# Without a credential, login fails before it saves anything.
isolate-home no_credential
env ANTITHESIS_API_KEY=
env ANTITHESIS_USERNAME=
env ANTITHESIS_PASSWORD=
! snouty login --tenant spec-tenant
stderr 'nothing was saved'
stderr -literal 'set ANTITHESIS_API_KEY, or run `snouty login` in a terminal'
! exists no_credential/.config/snouty/settings.toml

# The cases below need a registry response that staging cannot give.
[staging] skip

# A registry that takes the Docker login is saved, and login names the
# commands that sign in to it.
isolate-home docker_registry
mock-server 200 '{"auth":"docker","registry_url":"registry.example.com/acme"}'
set-env ANTITHESIS_API_KEY ${FAKE_API_KEY_A}
snouty login --tenant spec-tenant
stdout -literal '`docker login registry.example.com`'
stdout -literal '`podman login registry.example.com`'
file ~/.config/snouty/settings.toml 'repository = "registry.example.com/acme"'

# A registry login kind that snouty does not know gets no sign-in advice.
isolate-home unknown_registry
mock-server 200 '{"auth":"NEW_KIND","registry_url":"registry.example.com/acme"}'
snouty login --tenant spec-tenant
! stdout 'docker login'
file ~/.config/snouty/settings.toml 'repository = "registry.example.com/acme"'

# A credential that the tenant rejects fails the login, which saves nothing.
isolate-home rejected
mock-server 401 '{"message":"Invalid or expired bearer token."}'
! snouty login --tenant spec-tenant
stderr 'did not accept the credential; nothing was saved'
stderr -literal 'set ANTITHESIS_API_KEY to a valid key, or unset it'
! exists rejected/.config/snouty/settings.toml

# A GitHub Actions OIDC token that the tenant rejects keeps the GitHub Actions
# advice, because ANTITHESIS_API_KEY does not hold it.
mock-runs-server
env ANTITHESIS_API_KEY=
env ANTITHESIS_USERNAME=
env ANTITHESIS_PASSWORD=
isolate-home oidc_rejected
set-env GITHUB_ACTIONS true
set-env ACTIONS_ID_TOKEN_REQUEST_URL ${ANTITHESIS_BASE_URL}/github-actions/token?api-version=2.0
set-env ACTIONS_ID_TOKEN_REQUEST_TOKEN ${FAKE_OIDC_REQUEST_TOKEN}
! snouty login --tenant spec-tenant
stderr 'did not accept the credential; nothing was saved'
stderr 'id-token: write'
! stderr 'unset it'
