# snouty runs usage
#
# Show the usage hours of runs, newest first (`GET /api/v0/usage`). With
# --rollup, show the total of each day, week, or month instead. Times and
# periods use the local timezone.

# Like every runs command, usage requires an API key.
! snouty runs usage
stderr 'No Antithesis credentials found.*'

# --rollup takes day, week, or month.
! snouty runs usage --rollup year
stderr 'possible values: day, week, month'

# The mock usage entries fall on different days in UTC and in UTC+9, so a
# fixed timezone makes the expected times and periods exact.
env TZ=Asia/Tokyo

# Without --rollup, each run is one row: its run id, its hours, and the start
# and end of its usage in the local timezone. The command reads every page
# until the limit.
mock-runs-server
snouty runs usage
stdout '^(RUN ID +HOURS +STARTED +ENDED|No usage found\.)$'
[!staging] stdout '^usage-1 +10h 30m +2026-03-03 01:00:00 +2026-03-03 02:30:00$'
[!staging] stdout '^usage-3 +1h 12m +2026-03-01 05:00:00 +2026-03-01 05:15:00$'
[!staging] stdout '^usage-5 +100h 0m +2026-01-15 09:00:00 +2026-01-15 15:00:00$'
[!staging] stdout -count=6 '^.+$'

# The --json rows have the fields of the usage endpoint, with the times in
# RFC 3339 in the local timezone.
mock-runs-server
snouty --json runs usage -n 1
! stdout '^[^{]'
[!staging] stdout '^\{"run_id":"[^"]+","hours":[^,]+,"start_time":"[^"]+","end_time":"[^"]+"\}$'
[!staging] stdout '^\{"run_id":"usage-1","hours":10.5,"start_time":"2026-03-03T01:00:00\+09:00","end_time":"2026-03-03T02:30:00\+09:00"\}$'
[!staging] stdout -count=1 '^.+$'

# --limit caps the rows, and a note on stderr says that more can exist.
mock-runs-server
[!staging] snouty runs usage -n 2
[!staging] stdout '^usage-2 '
[!staging] ! stdout 'usage-3'
[!staging] stderr 'Showing up to 2 results'

# --started-after and --started-before filter on the start of the usage.
mock-runs-server
[!staging] snouty runs usage --started-after 2026-02-01T00:00:00Z --started-before 2026-03-01T12:00:00Z
[!staging] stdout '^usage-2 '
[!staging] stdout '^usage-4 '
[!staging] ! stdout 'usage-1'
[!staging] ! stdout 'usage-5'

# With --rollup, each row is one period: its last day, the number of runs
# that started in it, and the sum of their hours. Newest period first. The
# rows are the --limit periods that end with the current period, and a period
# with no runs shows 0.
mock-runs-server
snouty runs usage --rollup day
stdout '^DATE +RUNS +HOURS$'
stdout -count=11 '^.+$'
! stderr 'Showing up to'

# --started-before sets the newest period. HOURS is rounded to the minute:
# 2.1 + 1.2 hours shows as 3h 18m.
mock-runs-server
[!staging] snouty runs usage --rollup day -n 5 --started-before 2026-03-03T12:00:00Z
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-03 +1 +10h 30m\n2026-03-02 +0 +0h 0m\n2026-03-01 +2 +3h 18m\n2026-02-28 +0 +0h 0m\n2026-02-27 +1 +4h 0m$'
[!staging] stdout -count=6 '^.+$'

# A week starts on Monday and ends on Sunday.
mock-runs-server
[!staging] snouty runs usage --rollup week -n 3 --started-before 2026-03-03T12:00:00Z
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-08 +1 +10h 30m\n2026-03-01 +3 +7h 18m\n2026-02-22 +0 +0h 0m$'
[!staging] stdout -count=4 '^.+$'

# usage-3 started in February in UTC, but in March in the local timezone. The
# JSON hours of a period are rounded to hundredths.
mock-runs-server
snouty --json runs usage --rollup month -n 3
! stdout '^[^{]'
stdout -count=3 '^.+$'
mock-runs-server
[!staging] snouty --json runs usage --rollup month -n 3 --started-before 2026-03-03T12:00:00Z
[!staging] stdout '(?s)^\{"date":"2026-03-31","runs":3,"hours":13.8\}\n\{"date":"2026-02-28","runs":1,"hours":4.0\}\n\{"date":"2026-01-31","runs":1,"hours":100.0\}$'
[!staging] stdout -count=3 '^.+$'

# --started-after ends the window at the period that holds it, and that
# period counts only the runs inside the filter. The request asks only for
# runs in the window, so the old history is not read.
mock-runs-server
[!staging] snouty --verbose runs usage --rollup month -n 12 --started-after 2026-02-28T00:00:00Z --started-before 2026-03-03T12:00:00Z
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-31 +3 +13h 48m\n2026-02-28 +0 +0h 0m$'
[!staging] stdout -count=3 '^.+$'
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-28T00%3A00%3A00Z'
mock-runs-server
[!staging] snouty --verbose runs usage --rollup week -n 2 --started-before 2026-03-03T12:00:00Z
[!staging] stderr '^> GET .*/api/v0/usage\?.*start_time=2026-02-22T00%3A00%3A00Z'

# No usage prints a message, or no rows with --json. A rollup shows its
# periods with 0 runs.
[!staging] mock-runs-server empty
[!staging] snouty runs usage
[!staging] stdout '^No usage found.$'
[!staging] mock-runs-server empty
[!staging] snouty runs usage --rollup week -n 2 --started-before 2026-03-03T12:00:00Z
[!staging] stdout '(?s)^DATE +RUNS +HOURS\n2026-03-08 +0 +0h 0m\n2026-03-01 +0 +0h 0m$'
[!staging] mock-runs-server empty
[!staging] snouty --json runs usage
[!staging] ! stdout .
