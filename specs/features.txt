# Opt-in features
#
# A feature gates a command that isn't ready to be on for everyone — because
# the Antithesis API it calls is still changing shape, or because most tenants
# can't serve it yet. Gating lets such a command ship in a release without
# putting it in front of users who would only hit a wall.
#
# Features are enabled by id in SNOUTY_UNSTABLE_FEATURES, a comma-separated list. They
# are deliberately an environment variable and not a setting: the gate decides
# which subcommands the parser has, so it must be known before the command
# line is parsed, and a setting can't be read that early without first parsing
# --settings/--profile — which would mean parsing the command line twice.
#
# The gate mechanics are exercised through `runs-exec`, which gates
# `snouty runs exec`; `runs-search` (gating `snouty runs search` and the
# events-search routing of `snouty runs events`) rides the same machinery,
# covered in runs.txt.

# 1. With no features set, invoking `runs exec` says what is wrong and how to
#    fix it. Whoever types the command knows it exists, so the error names the
#    feature and the variable instead of pretending the command is absent.
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'is an unstable feature and is not enabled'
stderr 'SNOUTY_UNSTABLE_FEATURES=runs-exec'
stderr 'snouty runs exec --help'

# 2. It is absent from `runs --help`, so it isn't discoverable there.
snouty runs --help
stdout 'logs.*Stream moment logs'
! stdout '.*exec.*'

# 3. The gate hides the command; it does not remove it from the parser. So
#    `runs exec --help` still prints its help, which names the feature and how
#    to enable it. Someone has to already know the command to get here, and
#    what they find tells them the truth about it.
snouty runs exec --help
stdout 'gated behind the .runs-exec. unstable feature'
stdout 'SNOUTY_UNSTABLE_FEATURES=runs-exec'

# 4. Shell completions list it either way: clap_complete emits hidden
#    subcommands into the candidate list. Completing it while the feature is
#    off just leads to the error in step 1.
snouty completions bash
stdout 'logs exec events'

# 5. The variable turns it on: the command appears in help...
env SNOUTY_UNSTABLE_FEATURES=runs-exec
snouty runs --help
stdout 'exec.*Execute a command in a run'

# 6. ...and is callable. (It gets as far as needing credentials, which is
#    proof the gate let it through.)
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'No Antithesis credentials found.*'
! stderr 'unrecognized subcommand.*'

# 7. The list is comma-separated, and unknown ids are carried rather than
#    rejected: one exported SNOUTY_UNSTABLE_FEATURES is shared by every snouty on the
#    machine, so an id from a newer build must not break this one.
env SNOUTY_UNSTABLE_FEATURES=from-the-future,runs-exec
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'No Antithesis credentials found.*'

# 8. Whitespace and empty entries are ignored. (`set-env` keeps the spaces;
#    the `env` directive would split on them.)
set-env SNOUTY_UNSTABLE_FEATURES  runs-exec , ,
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'No Antithesis credentials found.*'

# 9. An id that isn't a feature this build knows enables nothing.
env SNOUTY_UNSTABLE_FEATURES=not-a-feature
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'is an unstable feature and is not enabled'

# 10. An empty value is "unset", not "a feature named the empty string".
env SNOUTY_UNSTABLE_FEATURES=
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'is an unstable feature and is not enabled'

# 11. A settings file cannot enable a feature — the key is not a setting, so
#     it is ignored like any other unknown key rather than erroring.
snouty --settings features.toml runs --help
! stdout '.*exec.*'

# 12. `snouty doctor` reports the enabled features, so what is on is visible
#     alongside the rest of the resolved configuration. doctor exits non-zero
#     here (no credentials or container runtime), but the table still prints.
env SNOUTY_UNSTABLE_FEATURES=runs-exec
! snouty doctor
stderr 'features.*runs-exec'

# 13. The row is omitted entirely when nothing is set — an empty row would be
#     noise on every ordinary run.
env SNOUTY_UNSTABLE_FEATURES=
! snouty doctor
! stderr 'features.*'

-- features.toml --
features = ["runs-exec"]
tenant = "from-file"
