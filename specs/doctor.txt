# snouty doctor
#
# Check environment configuration for Antithesis testing.

# Shows help text.
snouty doctor --help
stdout '.*Check environment configuration.*'
stdout '.*container runtime.*'
stdout '(?s)environment\s+variables'
stdout '.*--json.*'

# With nothing configured, doctor names the ways to get a credential the API
# commands accept — it must not suggest username/password, which is legacy auth
# (issue #145). The headline states the bare problem; the notes explain it.
! snouty doctor
stderr '.*No Antithesis credentials found.*'
stderr '.*needs credentials to authenticate.*'
stderr '.*ask Antithesis support.*'
! stderr '.*ANTITHESIS_USERNAME.*'

# --json emits a structured report on stdout and still exits non-zero when a
# required credential is missing (so CI can gate on it).
! snouty doctor --json
stdout '.*"ok": false.*'
stdout '.*"name": "credentials".*'
stdout '.*"status": "error".*'
stdout '.*"level": "error".*'

# With only username/password set, doctor reports one warning: the deprecated
# credentials work only for launch and debug. Exported credentials still hide
# whatever `snouty login` stores, so the next step also unsets them. (doctor
# fails here only because no tenant is set.)
env ANTITHESIS_USERNAME=testuser
env ANTITHESIS_PASSWORD=testpass
! snouty doctor
stderr '^  ⚠ username/password for user \[testuser\] only works for `snouty launch` and `snouty debug`$'
stderr '^      WARNING: username/password is deprecated, and `snouty runs` and other API commands refuse it$'
stderr '^      NOTE: read from the `ANTITHESIS_USERNAME` and `ANTITHESIS_PASSWORD` environment variables$'
stderr '^      NOTE: run `snouty login` to store an API key, then `unset ANTITHESIS_USERNAME ANTITHESIS_PASSWORD`$'
! stderr 'Using password credentials'
! stderr 'No credentials the API commands accept'
! stderr 'snouty login. to sign in'
# One source only, so doctor says nothing about a conflict.
! stderr '.*more than one credential source.*'

# --- API connectivity + version check (issue #146) ---
# Cases that fabricate a status via mock-server are [!staging]-gated: under
# SNOUTY_STAGING, mock-server passes through to the real API where fabricated
# responses don't hold. The 401 case is the exception — a bad key provokes the
# same status from the real gateway, so it runs in both modes. The --offline
# and unreachable cases use a controlled local setup and run in both modes. The
# mock + --offline cases assume a container runtime (CI provisions one); the
# unreachable case is robust either way.

# 2xx: doctor reports the live API and tenant release versions. The check runs
# only with an API key (the endpoint rejects basic auth), so set one on top of
# the basic creds mock-server configures.
[!staging] mock-server 200 '{"latest_api_version":"v1","release_version":"56.0"}'
[!staging] env ANTITHESIS_API_KEY=testkey
[!staging] snouty doctor
[!staging] stderr '.*Antithesis API reachable.*'
[!staging] stderr '.*latest API version: v1.*'
[!staging] stderr '.*tenant release version: 56.0.*'
# A tenant that old predates the APIs some commands need, so doctor warns for
# each one.
[!staging] stderr 'tenant serves the execute-command .container. field'
[!staging] stderr 'release 64.0\) — .runs exec. fails'

# 404: typically an older tenant predating the version endpoint, but possibly a
# proxy/route intercepting the request — either way it's still reachable and
# authenticated, so doctor stays green and warns without blaming the tenant.
[!staging] mock-server 404 '{}'
[!staging] env ANTITHESIS_API_KEY=testkey
[!staging] snouty doctor
[!staging] stderr '.*Antithesis API reachable.*'
[!staging] stderr '.*returned 404.*'
[!staging] stderr '.*proxy.*'

# 401 with an EMPTY body — exactly what the API gateway returns for a bad
# token (content-type: text/plain, zero length). The HTTP status is what
# classifies the failure, so doctor names the rejected credentials instead of
# reporting the API as unreachable because the body didn't parse (issue #180).
# This block runs against staging too: mock-server passes through to the real
# gateway, and the bad key below provokes a real rejection, so the issue #180
# scenario is exercised end to end rather than only against a mock of it. The
# mock answers 401; the live gateway answers 403 for a bad key (the probes on
# issue #180 recorded both). Either status is an authentication rejection, so
# the assertion accepts both. Against staging, the empty ANTITHESIS_BASE_URL
# sends the request to the tenant's own gateway: a proxy base URL can replace
# the bad key with a real credential, and then the gateway accepts it.
mock-server 401 ''
env ANTITHESIS_API_KEY=testkey
[staging] env ANTITHESIS_BASE_URL=
! snouty doctor
stderr '.*Antithesis API rejected authentication.*'
stderr '.*returned HTTP 40[13].*'
! stderr '.*Antithesis API unreachable.*'

# --offline skips the network check entirely: no version line, no API contact.
env ANTITHESIS_API_KEY=testkey
env ANTITHESIS_TENANT=testtenant
snouty doctor --offline
! stderr '.*Antithesis API.*'

# Connection failure: the check fails (non-zero) with a host-named note (the raw
# error detail follows it; its exact text is reqwest's and unit-tested instead).
env ANTITHESIS_API_KEY=testkey
env ANTITHESIS_TENANT=testtenant
env ANTITHESIS_BASE_URL=http://127.0.0.1:1
! snouty doctor
stderr '.*Antithesis API unreachable.*'
stderr '.*could not connect to 127.0.0.1.*'

# --- conflicting credential sources (issue #292) ---
# A credential exported in the environment hides one that `snouty login` stored
# in the credentials file. The scan reads no network, and without an API key
# doctor makes no API call, so this block needs no mock and no `--offline`.
isolate-home credential_conflict
env ANTITHESIS_API_KEY=
env ANTITHESIS_USERNAME=testuser
env ANTITHESIS_PASSWORD=testpass
env ANTITHESIS_TENANT=testtenant
# Doctor reports it as one check whose one next step frees the API key. It
# doesn't also report a credential shortfall and the password, and it doesn't
# tell the user to run `snouty login`, which they already did.
snouty doctor
stderr '^  ⚠ the username and password are hiding your API key$'
stderr '^      WARNING: snouty uses the username and password from the `ANTITHESIS_USERNAME` and `ANTITHESIS_PASSWORD` environment variables, which `snouty runs` refuses$'
stderr '^      NOTE: `unset ANTITHESIS_USERNAME ANTITHESIS_PASSWORD` to use the API key in .+credentials\.toml$'
! stderr 'snouty login. to sign in'
! stderr 'more than one credential source'
! stderr 'Using password credentials'

snouty doctor --json
stdout '"name": "credentials"'
stdout '"text": "`unset ANTITHESIS_USERNAME ANTITHESIS_PASSWORD` to use the API key in .*credentials\.toml"'
! stdout '"name": "credential_sources"'
! stdout '"name": "basic_auth"'
! stdout 'spec-stored-key|testpass'

# A username and password stored in the credentials file: `snouty login`
# replaces them, so it is the only next step. A warning does not fail doctor.
isolate-home stored_password
env ANTITHESIS_API_KEY=
env ANTITHESIS_USERNAME=
env ANTITHESIS_PASSWORD=
env ANTITHESIS_TENANT=testtenant
snouty doctor
stderr '^  ⚠ username/password for user \[storeduser\] only works for `snouty launch` and `snouty debug`$'
stderr '^      NOTE: read from the \[default\] profile in .+credentials\.toml$'
stderr '^      NOTE: run `snouty login` to replace it with an API key$'
! stderr 'unset ANTITHESIS'
! stderr 'Using password credentials'

snouty doctor --json
stdout '"name": "credentials",\s*"status": "warn"'
! stdout '"name": "basic_auth"'
! stdout 'stored-pass'

# A bare `snouty login` writes the default profile, so for a password stored
# under a named profile the next step names that profile.
snouty --profile prod doctor
stderr '^      NOTE: read from the \[prod\] profile in .+credentials\.toml$'
stderr '^      NOTE: run `snouty login --profile prod` to replace it with an API key$'

# A credentials file snouty cannot parse sits behind the exported API key. No
# command reads that far, so doctor reports the key it uses and stays green
# rather than failing on a file that stops nothing. `--offline` is needed here:
# the API key makes doctor call the API, and `testtenant` resolves nowhere.
isolate-home broken_credentials
env ANTITHESIS_API_KEY=testkey
env ANTITHESIS_TENANT=testtenant
env ANTITHESIS_USERNAME=
env ANTITHESIS_PASSWORD=
snouty doctor --offline
stderr '.*API key provided.*'
! stderr '(?s).*credentials.toml.*'
