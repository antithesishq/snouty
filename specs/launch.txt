# snouty launch
#
# Launch an Antithesis test run.
#
# As a developer, I want to launch an Antithesis test run from the command
# line so that I can trigger testing without using the web UI.
#
# Each successful invocation actually starts a test run against the API, so
# the bulk of this file is gated behind `[staging] skip`. The credential-free
# checks above the skip (arg parsing, schema validation) fail before any API
# call and run in both modes.

# -w / --webhook is a deprecated alias for --launcher. Help hides it, and it
#    conflicts with --launcher.
! stdout 'webhook'
! snouty launch --launcher basic_test --webhook basic_test --max-compute-hours 24
stderr 'cannot be used with'

# An invalid --duration fails at flag-parse time with a message attributed to
#    the flag, not an opaque schema regex.
! snouty launch --launcher basic_test --duration abc
stderr 'invalid value .abc. for .--duration'
stderr 'a bare number is not permitted'

# A --duration below 1 minute fails at flag-parse time, before any request,
#    so a typo such as `0` cannot start a run. A bare number that fails gets
#    no deprecation warning.
! snouty launch --launcher basic_test --config-image config:latest --duration 0
stderr 'invalid value .0. for .--duration <DURATION>.: must be at least 1 minute'
! stderr 'is deprecated; write'
! snouty launch --launcher basic_test --config-image config:latest --duration 0m
stderr 'must be at least 1 minute'
! snouty launch --launcher basic_test --config-image config:latest --duration 0h
stderr 'must be at least 1 minute'
! snouty launch --launcher basic_test --config-image config:latest --duration 59s
stderr 'must be at least 1 minute'

# --max-compute-hours must be a whole number, at most 1000. A value out of
#    range fails at flag-parse time, before any request.
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 0
stderr 'invalid value .0. for .--max-compute-hours <MAX_COMPUTE_HOURS>.: 0 is not in 1..=1000'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 1001
stderr '1001 is not in 1..=1000'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 48.5
stderr 'invalid value .48\.5. for .--max-compute-hours'

# --setup-timeout takes h/m units only. A bare number fails at flag-parse
#    time, because a reader cannot tell its unit. The timeout must be whole
#    minutes, from 1 to 30.
! snouty launch --launcher basic_test --config-image config:latest --setup-timeout 10
stderr 'invalid value .10. for .--setup-timeout <SETUP_TIMEOUT>.: must be a whole number with an h, m or s unit'
stderr 'a bare number is not permitted'
! snouty launch --launcher basic_test --config-image config:latest --setup-timeout 90s
stderr 'must be a whole number of minutes'
! snouty launch --launcher basic_test --config-image config:latest --setup-timeout 0m
stderr 'must be from 1 to 30 minutes'
! snouty launch --launcher basic_test --config-image config:latest --setup-timeout 30s
stderr 'must be from 1 to 30 minutes'
! snouty launch --launcher basic_test --config-image config:latest --setup-timeout 31m
stderr 'invalid value .31m. for .--setup-timeout <SETUP_TIMEOUT>.: must be from 1 to 30 minutes'
! snouty launch --launcher basic_test --config-image config:latest --setup-timeout 1h
stderr 'must be from 1 to 30 minutes'

# The budget must buy at least 15 minutes of fuzzing on the tier's cores:
#    standard 48, fast 96, turbo 192, so 12, 24 and 48 compute hours. The tier
#    defaults to standard, and the setup timeout does not change the minimum.
#    A budget below the minimum fails before any request, and the error names
#    the tier and the minimum.
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 11
stderr 'Error: --max-compute-hours 11 is too low for the standard tier: standard runs on 48 cores, so 15 minutes of fuzzing needs at least 12 compute hours'
stderr 'Suggestion: raise --max-compute-hours to at least 12'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 12 --performance-tier fast
stderr 'Error: --max-compute-hours 12 is too low for the fast tier: fast runs on 96 cores, so 15 minutes of fuzzing needs at least 24 compute hours'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 24 --performance-tier turbo
stderr 'too low for the turbo tier: turbo runs on 192 cores, so 15 minutes of fuzzing needs at least 48 compute hours'

# The same checks apply to values from --param. A setup timeout above 30
#    minutes fails. One below 1 minute passes, because the launcher raises it
#    to 1. A tier that this snouty does not know skips the budget check, and
#    the server decides.
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.max_compute_hours=11
stderr 'needs at least 12 compute hours'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 12 --param antithesis.performance_tier=fast
stderr 'too low for the fast tier'
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.setup_timeout=31
stderr 'Error: --setup-timeout 31 is above the maximum of 30 minutes'
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.setup_timeout=99999999999999999999999
stderr '--setup-timeout 99999999999999999999999 is above the maximum of 30 minutes'
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.setup_timeout=0
! stderr 'setup_timeout'
stderr 'No Antithesis credentials found'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 12 --param antithesis.performance_tier=ultra
! stderr 'too low'
stderr 'No Antithesis credentials found'

# --duration and --max-compute-hours cannot be used together.
! snouty launch --launcher basic_test --config-image config:latest --duration 30m --max-compute-hours 24
stderr 'cannot be used with'

# From --param, antithesis.max_compute_hours and antithesis.setup_timeout
#    must be whole numbers, and the schema check names a field that is not.
#    The schema does not check a range: the server sets it, and a tenant can
#    have a higher limit. So a value outside the flag's range reaches the
#    credentials check.
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.max_compute_hours=lots
stderr 'antithesis\.max_compute_hours: .*'
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.max_compute_hours=5000
! stderr 'validation failed'
stderr 'No Antithesis credentials found'
! snouty launch --launcher basic_test --config-image config:latest --param antithesis.setup_timeout=10m
stderr 'antithesis\.setup_timeout: .*'

# Parameters are validated against the testParams schema before any API
#    call. The error for a bad antithesis.* value names the field.
! snouty launch --launcher basic_test --param antithesis.duration=asdf --param antithesis.source=ci
stderr 'validation failed.*'
stderr 'antithesis\.duration: .*'
! stderr 'Unevaluated properties.*'

# `antithesis.filter_logs_matching` is validated client-side before any API
#    call. The platform rejects a bad pattern only after the run has started.
! snouty launch --launcher basic_test --max-compute-hours 24 --filter-logs-matching '('
stderr 'antithesis\.filter_logs_matching is not a valid regular expression.*'

# A pattern that matches the empty string would suppress every log line.
! snouty launch --launcher basic_test --max-compute-hours 24 --filter-logs-matching 'x*'
stderr 'antithesis\.filter_logs_matching matches the empty string, which would suppress every log line'

# An empty pattern makes the platform skip filtering silently.
! snouty launch --launcher basic_test --max-compute-hours 24 --param antithesis.filter_logs_matching=
stderr 'antithesis\.filter_logs_matching is empty'

# `antithesis.filter_source_matching` gets the same client-side checks.
! snouty launch --launcher basic_test --max-compute-hours 24 --filter-source-matching '('
stderr 'antithesis\.filter_source_matching is not a valid regular expression.*'
! snouty launch --launcher basic_test --max-compute-hours 24 --filter-source-matching '(foo)?'
stderr 'antithesis\.filter_source_matching matches the empty string, which would suppress the output of every source'
! snouty launch --launcher basic_test --max-compute-hours 24 --param antithesis.filter_source_matching=' '
stderr 'antithesis\.filter_source_matching is empty'

# --performance-tier accepts only the tiers in the vendored OpenAPI spec.
! snouty launch --launcher basic_test --max-compute-hours 24 --performance-tier ultra
stderr 'invalid value .ultra. for .--performance-tier'
stderr 'possible values: standard, fast, turbo'

# --fault-tolerance accepts only the levels in the vendored OpenAPI spec, by
#    their wire names.
! snouty launch --launcher basic_test --max-compute-hours 24 --fault-tolerance high-availability
stderr 'invalid value .high-availability. for .--fault-tolerance'
stderr 'possible values: none, resilient, high_availability, legacy'

# `antithesis.fault_exclusions` must be a non-empty, valid Rust regex, from
#    the flag or from --param.
! snouty launch --launcher basic_test --max-compute-hours 24 --fault-exclusions '(?=raft)'
stderr 'antithesis\.fault_exclusions is not a valid regular expression.*'
! snouty launch --launcher basic_test --max-compute-hours 24 --param antithesis.fault_exclusions='('
stderr 'antithesis\.fault_exclusions is not a valid regular expression.*'
! snouty launch --launcher basic_test --max-compute-hours 24 --fault-exclusions ''
stderr 'antithesis\.fault_exclusions is empty'

# `launch` does not accept --stdin.
! snouty launch --launcher basic_test --stdin --max-compute-hours 24
stderr 'unexpected argument.*'

# Old-style raw --antithesis.* args are not accepted by `launch`.
! snouty launch --launcher basic_test --antithesis.duration 30

# The basic_test and basic_k8s_test launchers need a config image. Without
#    --config, --config-image, or --param antithesis.config_image, the command
#    fails before any API call and names the flags that supply one.
! snouty launch --launcher basic_test --max-compute-hours 24
stderr 'the basic_test launcher requires a config image'
stderr 'pass --config or --config-image'
! snouty launch --launcher basic_k8s_test --max-compute-hours 24
stderr 'the basic_k8s_test launcher requires a config image'
! snouty launch --max-compute-hours 24
stderr 'the basic_test launcher requires a config image'

# The command authenticates with the Antithesis API using either
#    ANTITHESIS_API_KEY and ANTITHESIS_TENANT, or
#    ANTITHESIS_USERNAME, ANTITHESIS_PASSWORD, and ANTITHESIS_TENANT.
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 24
stderr 'No Antithesis credentials found.*'

# Everything below runs against the mock only. The skip sits above the first
# `mock-server` directive: under staging that directive installs the real
# tenant credentials, and no `snouty launch` command may execute with them —
# safety must not depend on where the command fails.
[staging] skip

# --param cannot override typed flag values.
mock-server 200 '{"message":"Success","runId":"run-123"}'
! snouty launch --launcher basic_test --max-compute-hours 24 --param antithesis.max_compute_hours=60
stderr 'cannot be overridden via --param.*'

# A --param value must be a key=value pair. snouty rejects any other form
#    before any API call.
! snouty launch --launcher basic_test --max-compute-hours 24 --param no-equals-sign
stderr 'invalid value .no-equals-sign. for .--param.*expected key=value'

# At least one source of parameters must be provided; otherwise the
#    command fails.
mock-server 200 '{}'
! snouty launch --launcher basic_test
stderr 'no parameters provided.*'

# Parameters are provided as --key value typed flags
#    (e.g. --max-compute-hours 24, --test-name my-test). Before sending, the
#    resolved parameters are printed to stderr with sensitive values (tokens, email
#    recipients) redacted. On success, a one-line confirmation with the
#    run_id is printed to stdout.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --test-name my-test --description 'nightly test run' --config-image config:latest --max-compute-hours 24 --recipients team@example.com --source ci-pipeline
stderr '"antithesis.test_name": "my-test"'
stderr '"antithesis.description": "nightly test run"'
stderr '"antithesis.config_image": "config:latest"'
stderr '"antithesis.max_compute_hours": "24"'
stderr '"antithesis.report.recipients": "\[REDACTED\]"'
stderr '"antithesis.source": "ci-pipeline"'
stderr '"run.caller_type": "snouty"'
stdout 'Test run started: run_id run-123'

# After the run ID, the output names the next steps: wait for the run to
#    finish, or interact with it once it is ready: wait for it first, then
#    follow its events, search them, or execute a script.
stdout '(?s)Test run started: run_id run-123\n\nWait for the run to finish with:\n  snouty runs wait run-123\n'
stdout '(?s)Or interact with the run once it is ready:\n  snouty runs wait run-123 --until ready\n  snouty runs search run-123 --follow <query>\n  snouty runs events run-123 <query>\n  snouty runs exec run-123 <input_hash> <vtime> <script>$'

# A --param value replaces the default run.caller_type.
mock-server 200 '{"message":"Success","runId":"run-123"}'

stderr '"run.caller_type": "ci"'
! stderr '"run.caller_type": "snouty"'

set-env ANTITHESIS_API_KEY ${FAKE_API_KEY_A}

# With --json, snouty prints its own envelope: the run id, and nothing else.
# The launch webhook's body-level statusCode is deliberately not re-exported
# (issue #180) — the exit code is the success signal, and the field disagrees
# with the response anyway (the mock answers HTTP 200 saying 202). Use --verbose
# to see the response as it arrived.
mock-server 200 '{"runId":"run-123","statusCode":202}'
snouty --json launch --launcher basic_test --config-image config:latest --max-compute-hours 24 --source ci-pipeline
stdout '\s*"runId": "run-123".*'

# 6c3. --duration is deprecated, and prints a warning that names
#    --max-compute-hours. It still sends `antithesis.duration`, and accepts h/m
#    unit suffixes, normalized to bare minutes for the API (1h30m -> 90).
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 1h30m --source ci-pipeline

# A 404 from the launch endpoint means that the launcher does not exist. Its
# body is the Launch_Error_Response envelope ({ statusCode, runId: null }),
# verified against the live API, with no message text. The command names the
# launcher and tells the user to ask support for the launchers on the tenant.
mock-server 404 '{"statusCode":404,"runId":null}'
! snouty launch --launcher no_such_launcher --max-compute-hours 24
stderr 'unknown launcher: no_such_launcher'
stderr 'ask support@antithesis\.com for the launchers on your tenant'
! stderr 'API error'

# A 404 without the envelope, such as an HTML page from a wrong host, keeps
# the generic API error, because it does not show that the launcher is unknown.
mock-server 404 '<html><body>404 Not Found</body></html>'
! snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 24
stderr 'API error: 404 Not Found'
! stderr 'unknown launcher'

# In a git checkout, the launch records the checked-out commit and branch as
#    `vcs.*` params. For a github.com `origin` remote, it also records the
#    repository and a link to the commit. The test ignores the global and
#    system git config, so that a local setting cannot change the commit.

env GIT_CONFIG_NOSYSTEM=1
exec git init -q -b feature/vcs checkout
cd checkout

# A pre-built config image has no config dir, and the current directory need
#    not hold its source, so it turns the detection off. `--param vcs.*`
#    values are still sent.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 24
! stderr 'vcs\.'
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --max-compute-hours 24 --param antithesis.config_image=config:latest
! stderr 'vcs\.'
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --max-compute-hours 24 --param vcs.version_id=abc123
stderr '"vcs.version_id": "abc123"'
! stderr 'vcs.repo_owner'

# A --param for a detected key turns the detection off, so that a detected
#    commit link cannot sit next to a different commit.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --max-compute-hours 24 --param vcs.version_id=abc123
stderr '"vcs.version_id": "abc123"'
! stderr 'vcs.version_link'
! stderr 'vcs.repo_branch'

# A --param for a `vcs.*` key that snouty does not detect is sent with the
#    detected values.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --max-compute-hours 24 --param vcs.system_name=widget
stderr '"vcs.system_name": "widget"'
stderr '"vcs.repo_owner": "acme"'
stderr '"vcs.version_id": "[0-9a-f]{40}"'

# Detected params and the default run.caller_type do not count as parameters
#    the user provided.
mock-server 200 '{}'
! snouty launch --launcher basic_test
stderr 'no parameters provided.*'

# --no-vcs turns the detection off.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --max-compute-hours 24 --no-vcs
! stderr 'vcs\.'

# A detached HEAD has no branch.
exec git checkout -q --detach
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --max-compute-hours 24
! stderr 'vcs.repo_branch'
stderr '"vcs.version_id": "[0-9a-f]{40}"'

# A GIT_DIR from a git hook does not redirect the detection.
env GIT_DIR=/nonexistent

snouty launch --launcher advanced_test --max-compute-hours 24
stderr '"vcs.repo_owner": "acme"'
