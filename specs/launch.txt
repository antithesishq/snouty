# snouty launch
#
# Launch an Antithesis test run.
#
# As a developer, I want to launch an Antithesis test run from the command
# line so that I can trigger testing without using the web UI.
#
# Each successful invocation actually starts a test run against the API, so
# the bulk of this file is gated behind `[staging] skip`. The credential-free
# checks above the skip (arg parsing, schema validation) fail before any API
# call and run in both modes.

# Environment variables are optional sources for settings and credentials.
snouty launch --help
stdout '^  ANTITHESIS_TENANT +Your Antithesis tenant name\.$'
stdout '^  ANTITHESIS_API_KEY +API key authentication\.$'
stdout '^  ANTITHESIS_USERNAME +Username \(deprecated\)\.$'
stdout '^  ANTITHESIS_PASSWORD +Password \(deprecated\)\.$'
stdout '^  ANTITHESIS_REPOSITORY +Container registry for pushing images\.$'
! stdout 'required when API key is not set'

# The help says where credentials come from and names the next step.
snouty launch --help
stdout '^Credentials come from `snouty login` or from the environment variables below\.$'
stdout 'wait for the run to finish with `snouty runs wait <run_id>`'
stdout '`snouty runs search <run_id> --follow <query>`'
stdout 'A pre-built\s+config image \(--config-image or --param antithesis\.config_image\) turns this\s+off'

# The help documents user-defined `attrs.*` params, which `runs show` lists.
snouty launch --help
stdout 'params with an `attrs\.` prefix'
stdout '--param attrs\.team=payments'

# The user specifies a launcher name via -l / --launcher. It defaults to
#    basic_test.
snouty launch --help
stdout '^  -l, --launcher <LAUNCHER>$'
stdout '\[default: basic_test\]'

# -w / --webhook is a deprecated alias for --launcher. Help hides it, and it
#    conflicts with --launcher.
! stdout 'webhook'
! snouty launch --launcher basic_test --webhook basic_test --duration 30
stderr 'cannot be used with'

# An invalid --duration fails at flag-parse time with a message attributed to
#    the flag, not an opaque schema regex.
! snouty launch --launcher basic_test --duration abc
stderr 'invalid value .abc. for .--duration'
stderr 'number of minutes.*'

# A --duration below 1 minute fails at flag-parse time, before any request,
#    so a typo such as `0` cannot start a run.
! snouty launch --launcher basic_test --config-image config:latest --duration 0
stderr 'invalid value .0. for .--duration <DURATION>.: must be at least 1 minute'
! snouty launch --launcher basic_test --config-image config:latest --duration 0m
stderr 'must be at least 1 minute'
! snouty launch --launcher basic_test --config-image config:latest --duration 0h
stderr 'must be at least 1 minute'
! snouty launch --launcher basic_test --config-image config:latest --duration 59s
stderr 'must be at least 1 minute'

# Parameters are validated against the testParams schema before any API
#    call. The error for a bad antithesis.* value names the field.
! snouty launch --launcher basic_test --param antithesis.duration=asdf --param antithesis.source=ci
stderr 'validation failed.*'
stderr 'antithesis\.duration: .*'
! stderr 'Unevaluated properties.*'

# `antithesis.filter_logs_matching` is validated client-side before any API
#    call. The platform rejects a bad pattern only after the run has started.
! snouty launch --launcher basic_test --duration 30 --filter-logs-matching '('
stderr 'antithesis\.filter_logs_matching is not a valid regular expression.*'

# A pattern that matches the empty string would suppress every log line.
! snouty launch --launcher basic_test --duration 30 --filter-logs-matching 'x*'
stderr 'antithesis\.filter_logs_matching matches the empty string, which would suppress every log line'

# An empty pattern makes the platform skip filtering silently.
! snouty launch --launcher basic_test --duration 30 --param antithesis.filter_logs_matching=
stderr 'antithesis\.filter_logs_matching is empty'

# `antithesis.filter_source_matching` gets the same client-side checks.
! snouty launch --launcher basic_test --duration 30 --filter-source-matching '('
stderr 'antithesis\.filter_source_matching is not a valid regular expression.*'
! snouty launch --launcher basic_test --duration 30 --filter-source-matching '(foo)?'
stderr 'antithesis\.filter_source_matching matches the empty string, which would suppress the output of every source'
! snouty launch --launcher basic_test --duration 30 --param antithesis.filter_source_matching=' '
stderr 'antithesis\.filter_source_matching is empty'

# --performance-tier accepts only the tiers in the vendored OpenAPI spec, and
#    its help points at --param for any other tier.
! snouty launch --launcher basic_test --duration 30 --performance-tier ultra
stderr 'invalid value .ultra. for .--performance-tier'
stderr 'possible values: standard, fast, turbo'
snouty launch --help
stdout 'possible values: standard, fast, turbo\]'
stdout '`--param=antithesis\.performance_tier=TIER`'

# `launch` does not accept --stdin.
! snouty launch --launcher basic_test --stdin --duration 30
stderr 'unexpected argument.*'

# Old-style raw --antithesis.* args are not accepted by `launch`.
! snouty launch --launcher basic_test --antithesis.duration 30

# The basic_test and basic_k8s_test launchers need a config image. Without
#    --config, --config-image, or --param antithesis.config_image, the command
#    fails before any API call and names the flags that supply one.
! snouty launch --launcher basic_test --duration 30
stderr 'the basic_test launcher requires a config image'
stderr 'pass --config or --config-image'
! snouty launch --launcher basic_k8s_test --duration 30
stderr 'the basic_k8s_test launcher requires a config image'
! snouty launch --duration 30
stderr 'the basic_test launcher requires a config image'

# The command authenticates with the Antithesis API using either
#    ANTITHESIS_API_KEY and ANTITHESIS_TENANT, or
#    ANTITHESIS_USERNAME, ANTITHESIS_PASSWORD, and ANTITHESIS_TENANT.
! snouty launch --launcher basic_test --config-image config:latest --duration 30
stderr 'No Antithesis credentials found.*'

# Everything below runs against the mock only. The skip sits above the first
# `mock-server` directive: under staging that directive installs the real
# tenant credentials, and no `snouty launch` command may execute with them —
# safety must not depend on where the command fails.
[staging] skip

# --param cannot override typed flag values.
mock-server 200 '{"message":"Success","runId":"run-123"}'
! snouty launch --launcher basic_test --duration 30 --param antithesis.duration=60
stderr 'cannot be overridden via --param.*'

# A --param value must be a key=value pair. snouty rejects any other form
#    before any API call.
! snouty launch --launcher basic_test --duration 30 --param no-equals-sign
stderr 'invalid value .no-equals-sign. for .--param.*expected key=value'

# At least one source of parameters must be provided; otherwise the
#    command fails.
mock-server 200 '{}'
! snouty launch --launcher basic_test
stderr 'no parameters provided.*'

# Parameters are provided as --key value typed flags
#    (e.g. --duration 30, --test-name my-test). Before sending, the resolved
#    parameters are printed to stderr with sensitive values (tokens, email
#    recipients) redacted. On success, a one-line confirmation with the
#    run_id is printed to stdout.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --test-name my-test --description 'nightly test run' --config-image config:latest --duration 30 --recipients team@example.com --source ci-pipeline
stderr '"antithesis.test_name": "my-test"'
stderr '"antithesis.description": "nightly test run"'
stderr '"antithesis.config_image": "config:latest"'
stderr '"antithesis.duration": "30"'
stderr '"antithesis.report.recipients": "\[REDACTED\]"'
stderr '"antithesis.source": "ci-pipeline"'
stderr '"run.caller_type": "snouty"'
stdout 'Test run started: run_id run-123'

# After the run ID, the output names the next steps: wait for the run to
#    finish, or watch its errors live with `runs search --follow`. A live
#    search needs a running run, so that step waits for it first.
stdout '(?s)Test run started: run_id run-123\n\nWait for the run to finish with:\n  snouty runs wait run-123\n'
stdout '(?s)Or watch its errors live once it is ready:\n  snouty runs wait run-123 --until ready\n  snouty runs search run-123 --follow .contains\(.output_text: "error".\).$'

# A --param value replaces the default run.caller_type.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --param run.caller_type=ci
stderr '"run.caller_type": "ci"'
! stderr '"run.caller_type": "snouty"'

# Username/password authentication still works for the launch webhook, but it
# is deprecated: a warning on stderr steers the user to `snouty login`. The
# mock-server directive configures username/password credentials.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --source ci-pipeline
stderr 'warning: username/password authentication is deprecated'
stderr 'snouty login'
stdout 'Test run started: run_id run-123'

# With an API key configured, no deprecation warning is shown. (An empty env
# value means unset, so the last line restores username/password auth for the
# rest of the file.)
mock-server 200 '{"message":"Success","runId":"run-123"}'
env ANTITHESIS_API_KEY=testkey
snouty launch --launcher basic_test --config-image config:latest --duration 30 --source ci-pipeline
! stderr 'deprecated'
stdout 'Test run started: run_id run-123'
env ANTITHESIS_API_KEY=

# With --json, snouty prints its own envelope: the run id, and nothing else.
# The launch webhook's body-level statusCode is deliberately not re-exported
# (issue #180) — the exit code is the success signal, and the field disagrees
# with the response anyway (the mock answers HTTP 200 saying 202). Use --verbose
# to see the response as it arrived.
mock-server 200 '{"runId":"run-123","statusCode":202}'
snouty --json launch --launcher basic_test --config-image config:latest --duration 30 --source ci-pipeline
stdout '\s*"runId": "run-123".*'
! stdout '.*statusCode.*'
! stdout 'Test run started.*'
! stdout 'snouty runs'

# 6b. --ephemeral flag sets antithesis.is_ephemeral to "true".
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --ephemeral
stderr '"antithesis.is_ephemeral": "true"'

# 6c. Without --source, is_ephemeral is set automatically and an info message is shown.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30
stderr '"antithesis.is_ephemeral": "true"'
stderr 'Starting an ephemeral run; its findings will not be retained.*'

# 6c2. With --source, is_ephemeral is not set unless --ephemeral is passed.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --source ci-pipeline
! stderr 'is_ephemeral.*'
! stderr 'Starting an ephemeral run.*'

# 6c3. --duration accepts h/m unit suffixes, normalized to bare minutes for the
#    API (1h30m -> 90).
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 1h30m --source ci-pipeline
stderr '"antithesis.duration": "90"'

# 6d. --param flag passes custom properties.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --param my.custom.prop=value --param antithesis.integrations.github.callback_url=https://example.com/cb
stderr '"my.custom.prop": "value"'
stderr '"antithesis.integrations.github.callback_url": "https://example.com/cb"'

# 6d2. Unknown antithesis.* params pass through: the platform can add params
#    before snouty learns about them (#211). Known params keep their value
#    checks (see the antithesis.duration=asdf case above).
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --param antithesis.brand_new_param=on
stderr '"antithesis.brand_new_param": "on"'
stdout 'Test run started: run_id run-123'

# 6e. antithesis.images may be passed via --param to register additional
#    images (e.g. an image referenced only in a CRD field) that the config
#    parser can't discover. Semicolon-delimited [REGISTRY/]NAME(:TAG|@DIGEST).
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --param 'antithesis.images=app@sha256:abc;db:latest'
stderr '"antithesis.images": "app@sha256:abc;db:latest"'

# A 404 from the launch endpoint means that the launcher does not exist. Its
# body is the Launch_Error_Response envelope ({ statusCode, runId: null }),
# verified against the live API, with no message text. The command names the
# launcher and tells the user to ask support for the launchers on the tenant.
mock-server 404 '{"statusCode":404,"runId":null}'
! snouty launch --launcher no_such_launcher --duration 30
stderr 'unknown launcher: no_such_launcher'
stderr 'ask support@antithesis\.com for the launchers on your tenant'
! stderr 'API error'

# A 404 without the envelope, such as an HTML page from a wrong host, keeps
# the generic API error, because it does not show that the launcher is unknown.
mock-server 404 '<html><body>404 Not Found</body></html>'
! snouty launch --launcher basic_test --config-image config:latest --duration 30
stderr 'API error: 404 Not Found'
! stderr 'unknown launcher'

# Other errors that reach the test launcher show the HTTP status and the body.
# (The launcher does no request validation beyond auth, so it never returns a
# 400.) TODO: test a 400 here once the launcher validates requests.
mock-server 500 '{"statusCode":500,"runId":null}'
! snouty launch --launcher basic_test --config-image config:latest --duration 30
stderr 'API error: 500 Internal Server Error.*'

# `snouty run` still works but prints a deprecation warning.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty run --launcher basic_test --config-image config:latest --duration 30
stderr 'snouty run.*is deprecated.*snouty launch'

# Without --launcher, the run goes to the basic_test launcher.

# In a git checkout, the launch records the checked-out commit and branch as
#    `vcs.*` params. For a github.com `origin` remote, it also records the
#    repository and a link to the commit. The test ignores the global and
#    system git config, so that a local setting cannot change the commit.
env GIT_CONFIG_GLOBAL=/dev/null
env GIT_CONFIG_NOSYSTEM=1
exec git init -q -b feature/vcs checkout
cd checkout
exec git -c user.name=snouty -c user.email=snouty@example.com commit -q --allow-empty -m init
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --duration 30
stderr '"vcs.repo_branch": "feature/vcs"'
stderr '"vcs.version_id": "[0-9a-f]{40}"'
! stderr 'vcs.repo_type'
! stderr 'vcs.version_link'

exec git remote add origin git@github.com:acme/widget.git
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --duration 30
stderr '"vcs.repo_type": "github"'
stderr '"vcs.repo_owner": "acme"'
stderr '"vcs.repo_name": "widget"'
stderr '"vcs.repo_branch": "feature/vcs"'
stderr '"vcs.version_id": "[0-9a-f]{40}"'
stderr '"vcs.version_link": "https://github.com/acme/widget/commit/[0-9a-f]{40}"'

# A pre-built config image has no config dir, and the current directory need
#    not hold its source, so it turns the detection off. `--param vcs.*`
#    values are still sent.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30
! stderr 'vcs\.'
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --duration 30 --param antithesis.config_image=config:latest
! stderr 'vcs\.'
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher basic_test --config-image config:latest --duration 30 --param vcs.version_id=abc123
stderr '"vcs.version_id": "abc123"'
! stderr 'vcs.repo_owner'

# A --param for a detected key turns the detection off, so that a detected
#    commit link cannot sit next to a different commit.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --duration 30 --param vcs.version_id=abc123
stderr '"vcs.version_id": "abc123"'
! stderr 'vcs.version_link'
! stderr 'vcs.repo_branch'

# A --param for a `vcs.*` key that snouty does not detect is sent with the
#    detected values.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty launch --launcher advanced_test --duration 30 --param vcs.system_name=widget
stderr '"vcs.system_name": "widget"'
stderr '"vcs.repo_owner": "acme"'
stderr '"vcs.version_id": "[0-9a-f]{40}"'

# Detected params and the default run.caller_type do not count as parameters
