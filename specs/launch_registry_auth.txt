# snouty launch --config and the tenant registry
#
# When the repository is on the tenant registry,
# registry.<tenant>.antithesis.com, snouty gives the registry secret to the
# container runtime when it pushes. The secret is an API key, a GitHub Actions
# OIDC token, or an OAuth token, with the username snouty-token. The registry
# does not accept a username and password, so with it snouty gives no
# credential, and the runtime pushes with its own credentials, as for any
# other registry.
#
# Docker reads the credential from DOCKER_AUTH_CONFIG. Only Docker 28.3.0 and
# later read that variable, and an older Docker pushes with the credentials
# of its own config.
#
# Podman never gets the secret in a file: it gets a temporary auth file that
# sends it to the docker-credential-snouty helper for the registry. When the
# helper is not on PATH, snouty links it to itself in the global settings
# dir, which holds the credentials and which only the user can write, and
# puts that dir first on the PATH of podman. snouty writes the link only
# when it does not point at snouty. The helper gives the credential of the
# settings of the launch: it runs in the working directory of the launch,
# and the launch gives it its --settings and --profile.
#
# The fake engine below gives the --version banner of its engine, builds, and
# fails each other command. Its push fails with its arguments, which
# credential variables it got, the variables that select the settings, and an
# auth error, unless STUB_PUSH_OK is set. When REGISTRY_AUTH_FILE is set, the
# push also shows the auth file, the first directory of PATH and its files,
# and what that directory's docker-credential-snouty gives for the registry
# of the auth file, as podman runs it.

-- bin/engine --
#!/bin/sh
engine=$(basename "$0")
case "$1" in
  --version)
    case "$engine" in
      podman) echo 'podman version 4.9.3';;
      *) echo 'Docker version 28.3.0, build 38b7060';;
    esac;;
  build) cat >/dev/null;;
  push)
    if [ -n "$STUB_PUSH_OK" ]; then
      for arg in "$@"; do case "$arg" in --digestfile=*) printf 'sha256:pushed' > "${arg#--digestfile=}";; esac; done
      echo 'latest: digest: sha256:pushed size: 1'
      exit 0
    fi
    echo "fake $engine: $* REGISTRY_AUTH_FILE=[${REGISTRY_AUTH_FILE:+set}] DOCKER_AUTH_CONFIG=[${DOCKER_AUTH_CONFIG:+set}]" >&2
    echo "settings: profile=[${ANTITHESIS_PROFILE-}] path=[${SNOUTY_SETTINGS_PATH-}]" >&2
    if [ -n "$REGISTRY_AUTH_FILE" ]; then
      echo "auth file: $REGISTRY_AUTH_FILE $(cat "$REGISTRY_AUTH_FILE")" >&2
      helper_dir=${PATH%%:*}
      echo "helper dir: $helper_dir" $(ls "$helper_dir") >&2
      host=$(sed -n 's/.*"credHelpers":{"\([^"]*\)".*/\1/p' "$REGISTRY_AUTH_FILE")
      credential=$(printf %s "$host" | "$helper_dir/docker-credential-snouty" get)
      echo "helper: $credential" >&2
    fi
    echo 'unauthorized: authentication required' >&2
    exit 1;;
  *) echo "fake $engine: $1" >&2; exit 1;;
esac
