# snouty validate
#
# Validate a local Antithesis setup by running docker-compose and
# watching for the setup-complete event.

# Requires a config directory argument.
! snouty validate
stderr 'required.*'

# Rejects a nonexistent directory.
! snouty validate /nonexistent/path
stderr 'not a directory.*'

# Rejects a directory without docker-compose.yaml or manifests/.
! snouty validate empty
stderr '.*does not contain.*docker-compose.yaml.*manifests/.*'

# Lowercase docker-compose.yml is rejected with a rename hint.
! snouty validate yml_dir
stderr '.*rename.*'

# Rejects an ambiguous directory containing both compose and manifests/.
! snouty validate ambiguous
stderr '.*both docker-compose.yaml and a manifests/ subdirectory.*'

# h/m/s units parse. The command then fails on the missing directory, not on
# the flag.
! snouty validate /nonexistent/path --timeout 90s
stderr 'not a directory.*'
! stderr 'is deprecated; write'
! snouty validate /nonexistent/path --timeout 2m
stderr 'not a directory.*'
! stderr 'is deprecated; write'

# A bare number still counts seconds, but is deprecated: a warning shows the
# value with a unit.
! snouty validate /nonexistent/path --timeout 90
stderr '^warning: a bare number for --timeout is deprecated; write 1m30s$'
stderr 'not a directory.*'

# A value that is not h/m/s units fails, and the error gives the format.
! snouty validate empty --timeout 2x
stderr 'invalid value .2x. for .--timeout <TIMEOUT>.: must be a whole number with an h, m or s unit, or several in that order \(e\.g\. `90s`, `5m`, `1h30m`\); a bare number is not permitted'
! snouty validate empty --timeout 1.5
stderr 'a bare number is not permitted'
! stderr 'is deprecated; write'
