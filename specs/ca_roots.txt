# CA roots
#
# When no system CA certificates load, snouty trusts the bundled Mozilla
# roots. On Linux, an empty SSL_CERT_FILE makes the system load no roots.

[!linux] skip

# With no system CA roots, a command runs and warns about the empty variable.
env SSL_CERT_FILE=
mock-runs-server
snouty runs
stdout '^RUN ID +STATUS +CREATED +LAUNCHER +RUN$'
stderr '^warning: SSL_CERT_FILE or SSL_CERT_DIR loads no CA certificates'
