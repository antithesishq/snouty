# snouty runs wait
#
# Wait until a run is ready or complete (`--until`, default complete), then
# report its status. Complete means a terminal state: completed, cancelled, or
# incomplete. The command exits 0 for every status it stops at; the run's
# outcome is in the output, not the exit code. A run that reports status
# `unknown` fails the command instead of waiting forever on a run that may
# never make progress.
#
# Status transitions mid-wait are covered by Rust integration tests in
# src/runs.rs, where the test server can change the run's status between polls.

# Like every runs command, wait requires an API key.
! snouty runs wait some-run
stderr 'No Antithesis credentials found.*'

# A run ID is required.
! snouty runs wait
stderr '.*RUN_ID.*'

# The poll interval takes the same duration forms as `launch --duration`
# (minutes, or h/m/s units) and has a 1-minute floor: polling faster cannot
# observe a run (which takes minutes to hours) any sooner.
! snouty runs wait some-run --poll-interval 30s
stderr 'poll interval must be at least 1 minute'
! snouty runs wait some-run --poll-interval abc
stderr 'must be a number of minutes'

# `snouty runs --help` lists the wait subcommand.
snouty runs --help
stdout 'wait.*ready or complete'

# `snouty runs wait --help` explains the two --until statuses, that each is a
# minimum, the unknown-status failure, and that re-running resumes the wait
# after an interruption.
snouty runs wait --help
stdout 'completed, cancelled, or incomplete'
stdout 'unknown'
stdout 'interrupt and re-run'
stdout '--poll-interval'
stdout '--timeout'
stdout '--until'
stdout 'ready .*ready to use, or it is'
stdout 'accepts commands'
stdout 'from before release 64, is ready while it is in_progress'
stdout 'The status is a minimum'
! stdout 'runs exec'

# A run that is already terminal returns immediately: the final status is the
# whole output, exit 0.
mock-runs-server
snouty --json runs list --status completed
env_from_json 0 run_id

mock-runs-server
snouty runs wait $R_run_id
stdout 'run .* is completed'
! stdout 'Launcher'

# `snouty --json runs wait` reports the run id and final status as JSON.
mock-runs-server
snouty --json runs wait $R_run_id
stdout '"run_id".*"$R_run_id"'
[!staging] stdout '"status".*"completed"'

# incomplete is a terminal state too: the wait succeeds and reports it.
# run-3 is the mock's incomplete fixture.
mock-runs-server
[!staging] snouty runs wait run-3
[!staging] stdout 'run run-3 is incomplete'

# A run that reports status `unknown` fails the wait: snouty cannot tell
# whether such a run will still make progress, so the caller decides what to
# do. The error names the status the server sent and points at `runs show`.
mock-runs-server
[!staging] ! snouty runs wait run-unknown-status
[!staging] stderr 'run run-unknown-status reported status "unknown"'
[!staging] stderr 'snouty runs show run-unknown-status'

# A bad run id gives the same friendly "run not found" message as every other
# run-scoped subcommand.
mock-runs-server
! snouty runs wait no-such-run
stderr '.*run not found: no-such-run.*'
! stderr '.*API error.*'

# --timeout bounds the whole wait, in-flight request included. run-2 stays
# in_progress on the mock, so a zero timeout fails immediately.
mock-runs-server
[!staging] ! snouty runs wait run-2 --timeout 0
[!staging] stderr 'timed out waiting for run run-2'

# --until takes one of two points in the run's lifecycle.
! snouty runs wait some-run --until in_progress
stderr 'possible values: ready, complete'
! snouty runs wait some-run --until running
stderr 'possible values: ready, complete'

# `--until ready` stops once the run is ready, or complete. A completed run
# never becomes ready, so the wait succeeds at once and reports the run's
# actual status. A complete run has no sub-status, so the JSON has none.
mock-runs-server
snouty runs wait $R_run_id --until ready
stdout 'run .* is completed$'

mock-runs-server
snouty --json runs wait $R_run_id --until ready
[!staging] stdout '"status".*"completed"'
! stdout 'sub_status'

# run-2 is in_progress and ready on the mock, so the wait ends. The timeout
# only keeps a regression from hanging the spec.
mock-runs-server
[!staging] snouty runs wait run-2 --until ready --timeout 10s
[!staging] stdout '^run run-2 is in_progress:ready$'

mock-runs-server
[!staging] snouty --json runs wait run-2 --until ready --timeout 10s
[!staging] stdout '"sub_status": "Ready"'

# A debugging session in `in_progress` is not ready until it accepts
# commands, so the wait goes on. The first poll is at once, and the next one
# is a minute later, so a wait that stops at the first poll returns before
# the timeout.
mock-runs-server
[!staging] ! snouty runs wait session-loading --until ready --timeout 3s
[!staging] stderr 'timed out waiting for run session-loading'

# `--until complete` is the default: it waits for a terminal state.
mock-runs-server
[!staging] ! snouty runs wait run-2 --until complete --timeout 0
[!staging] stderr 'timed out waiting for run run-2'
