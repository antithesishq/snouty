# snouty docker-credential: the Docker credential helper protocol for the
# tenant registry, registry.<tenant>.antithesis.com.
#
# Docker writes the server URL on stdin and reads JSON from stdout. The
# username is always snouty-token, and the secret is the Antithesis
# credential that every command uses. Settings resolve as for every command:
# the environment, then the file that --settings or SNOUTY_SETTINGS_PATH
# names, then the global settings file. When the tenant registry of the
# settings is not the requested host, the credential is not found.
#
# A host that is not registry.<tenant>.antithesis.com is not found, and the
# helper reads no settings for it.

# The settings and the credentials in this file are local fixtures.
[staging] skip

env RUST_LOG=

# The global settings file uses registry.acme.antithesis.com, and the stored
# API key serves it.
isolate-home stored
substitute ~/.config/snouty/credentials.toml
stdin acme.txt
snouty docker-credential get
stdout -count=1 '^\{"ServerURL":"registry.acme.antithesis.com","Username":"snouty-token","Secret":"${FAKE_API_KEY_A@R}"\}$'
! stderr .

# A scheme, a path and capital letters do not change the host. ServerURL
# echoes the request as Docker gave it.
stdin acme_url.txt
snouty docker-credential get
stdout '^\{"ServerURL":"https://Registry.acme.antithesis.com/v2/","Username":"snouty-token","Secret":"${FAKE_API_KEY_A@R}"\}$'

# Another tenant registry is not found: the exact "not found" message, exit
# status 1, and no noise on stderr.
stdin beta.txt
! snouty docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stdout '\{'
! stderr .

# A host that is not a tenant registry is not found either.
stdin elsewhere.txt
! snouty docker-credential get
stdout '^credentials not found in native keychain$'

# snouty reads no ./.snouty.toml.
cd project
stdin omega.txt
! snouty docker-credential get
stdout '^credentials not found in native keychain$'
cd .

# SNOUTY_SETTINGS_PATH or --settings names a settings file, which overrides
# the global settings file.
env SNOUTY_SETTINGS_PATH=explicit.toml
stdin zeta.txt
snouty docker-credential get
stdout '"ServerURL":"registry.zeta.antithesis.com".*"Secret":"${FAKE_API_KEY_A@R}"'
stdin acme.txt
! snouty docker-credential get
stdout '^credentials not found in native keychain$'
env SNOUTY_SETTINGS_PATH=

stdin zeta.txt
snouty --settings explicit.toml docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'

# A settings file that it names must exist.
env SNOUTY_SETTINGS_PATH=missing.toml
stdin acme.txt
! snouty docker-credential get
stdout 'missing.toml.* was not found$'
env SNOUTY_SETTINGS_PATH=

# A settings value that does not resolve fails the helper. The error goes to
# stdout, where Docker shows it, and to stderr.
stdin acme.txt
! snouty --settings broken.toml docker-credential get
stdout '^invalid setting `tenant` in .*: `not a host` is not a valid tenant'
stderr '`not a host` is not a valid tenant'
! stdout '"Secret"'

# A host that is not a tenant registry is not found before the helper reads
# the settings, so the broken file is no error.
stdin elsewhere.txt
! snouty --settings broken.toml docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stderr .

# ANTITHESIS_API_KEY wins over a stored credential, as for every command.
set-env ANTITHESIS_API_KEY ${FAKE_API_KEY_B}
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'

# Whitespace around ANTITHESIS_API_KEY is not part of the key.
set-env ANTITHESIS_API_KEY " ${FAKE_API_KEY_B}\n"
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'

# An ANTITHESIS_API_KEY of only whitespace fails, and names the variable.
set-env ANTITHESIS_API_KEY " \n"
stdin acme.txt
! snouty docker-credential get
stdout 'ANTITHESIS_API_KEY is blank'
env ANTITHESIS_API_KEY=

# Environment settings override the files.
env ANTITHESIS_TENANT=delta
env ANTITHESIS_REGISTRY=registry.delta.antithesis.com/app
stdin delta.txt
snouty docker-credential get
stdout '"ServerURL":"registry.delta.antithesis.com".*"Secret":"${FAKE_API_KEY_A@R}"'
env ANTITHESIS_TENANT=
env ANTITHESIS_REGISTRY=

# A request with no host fails.
stdin blank.txt
! snouty docker-credential get
stdout '^invalid server URL on stdin: `` is not a container registry: give a host'

# store and list fail with one line on stdout, where the engine shows it.
# Docker and Podman call store for a login, and snouty owns the credential.
# Neither engine calls list for a "credHelpers" entry.
substitute store.json
stdin store.json
! snouty docker-credential store
stdout -count=1 '^snouty provides credentials for this registry; to sign in, run `snouty login` or set ANTITHESIS_API_KEY$'
! snouty docker-credential list
stdout -count=1 '^the snouty credential helper is not a general-purpose credential store; use it only as a "credHelpers" entry for registry.<tenant>.antithesis.com, not as "credsStore"$'

# erase reads stdin, erases nothing, and succeeds, because a failed erase
# breaks `podman logout --all` for every registry.
stdin acme.txt
snouty docker-credential erase
! stdout .

# They change nothing: get still serves the stored credential.
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'

# An action outside the protocol fails as a usage error.
! snouty docker-credential fetch
stderr 'invalid value .fetch.'

# Without a credential, the helper fails.
isolate-home no_credential
stdin nu.txt
! snouty docker-credential get
stdout -count=1 '^No Antithesis credentials found$'
stderr 'run `snouty login` or set ANTITHESIS_API_KEY'

# In a GitHub Actions job, the helper names the OIDC permission instead.
env GITHUB_ACTIONS=true
stdin nu.txt
! snouty docker-credential get
stderr -literal 'add `id-token: write` to the job\'s `permissions`'
env GITHUB_ACTIONS=

# OAuth: get refreshes a token whose expiry it cannot read, and stores the
# new token. A token that expires in more than 2 hours is used with no
# refresh. The tests in src/registry.rs cover the expiry margin.
mock-runs-server
env ANTITHESIS_API_KEY=
env ANTITHESIS_REGISTRY=registry.testtenant.antithesis.com/app
isolate-home oauth
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_B@R}"'
file ~/.config/snouty/credentials.toml 'antithesis_token = "${FAKE_ACCESS_TOKEN_B@R}"'
file ~/.config/snouty/credentials.toml 'refresh_token = "${FAKE_REFRESH_TOKEN_A@R}"'

# When the refresh fails, the stored token serves while it has not expired,
# with a warning that does not show the token.
env ANTITHESIS_BASE_URL=http://127.0.0.1:9
isolate-home oauth_refresh_fails
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_A@R}"'
stderr '^warning: the sign-in could not be renewed, so the stored token is used until it expires; '
! stderr '${FAKE_ACCESS_TOKEN_A@R}'

# When the stored token has expired too, the refresh error is the result.
isolate-home oauth_expired_refresh
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
! snouty docker-credential get
stdout '^the tenant could not renew the sign-in; try again later'
! stdout '"Secret"'
! stderr 'warning:'
env ANTITHESIS_BASE_URL=

# Without a refresh token, the stored token serves while it has not expired,
# with a warning.
isolate-home oauth_no_refresh
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_A@R}"'
stderr '^warning: the sign-in could not be renewed, so the stored token is used until it expires; '
! stderr 'expired and'

# An expired token without a refresh token fails, and names the fix, with no
# backtrace.
isolate-home oauth_expired
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
env RUST_BACKTRACE=1
! snouty docker-credential get
stdout '^the sign-in expired and could not be renewed; run `snouty login` to sign in again$'
stderr 'run `snouty login`'
! stderr 'BACKTRACE'
env RUST_BACKTRACE=

# A username and password fail, with the fix.
isolate-home password
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
! snouty docker-credential get
stdout 'does not accept username/password'

# In a GitHub Actions workflow, the secret is the workflow's OIDC token,
# which snouty gets from the Actions token endpoint.
mock-runs-server
env ANTITHESIS_API_KEY=
isolate-home oidc
set-env ACTIONS_ID_TOKEN_REQUEST_URL ${ANTITHESIS_BASE_URL}/github-actions/token?api-version=2.0
set-env ACTIONS_ID_TOKEN_REQUEST_TOKEN ${FAKE_OIDC_REQUEST_TOKEN}
stdin testtenant.txt
snouty docker-credential get
stdout -count=1 '^\{"ServerURL":"registry.testtenant.antithesis.com","Username":"snouty-token","Secret":"${FAKE_OIDC_TOKEN@R}"\}$'

# A failed token request fails the helper.
set-env ACTIONS_ID_TOKEN_REQUEST_URL ${ANTITHESIS_BASE_URL}/no-such-endpoint?api-version=2.0
stdin testtenant.txt
! snouty docker-credential get
stdout 'failed to get a GitHub Actions OIDC token: .*answered [0-9]{3} [^:]*: [^{]'
! stdout '"Secret"'
