# snouty debug
#
# Launch an Antithesis multiverse debugging session.
#
# As a developer, I want to launch a debugging session from the command line
# using details from a triage report so that I can reproduce and investigate
# a specific moment in time.
#
# Each successful invocation actually starts a debugging session against the
# API, so the bulk of this file is gated behind `[staging] skip` to avoid
# mutating staging. The skip sits above the first `mock-server` directive:
# under staging that directive installs the real tenant credentials, and no
# `snouty debug` command may execute with them — safety must not depend on
# where the command fails. Only the credential-free check above the skip runs
# in both modes.

# The command authenticates with the Antithesis API using either
#    ANTITHESIS_API_KEY and ANTITHESIS_TENANT, or
#    ANTITHESIS_USERNAME, ANTITHESIS_PASSWORD, and ANTITHESIS_TENANT.
! snouty debug --input-hash abc --run-id run-123 --vtime 123
stderr 'No Antithesis credentials found.*'

# The help documents --param for custom parameters.
snouty debug --help
stdout '--param attrs\.team=payments'

# A --param value must be a key=value pair. snouty rejects any other form
#    before any API call.
! snouty debug --input-hash abc --run-id run-123 --vtime 123 --param no-equals-sign
stderr 'invalid value .no-equals-sign. for .--param.*expected key=value'
! snouty debug --input-hash abc --run-id run-123 --vtime 123 --param =value
stderr 'invalid value .=value. for .--param.*the key is empty'

[staging] skip

# Parameters are validated against the debuggingParams schema before
#    making any API call. Required fields: input_hash, vtime, and exactly
#    one of run_id or session_id.
mock-server 200 '{}'
! snouty debug --input-hash abc
stderr 'validation failed.*'

# --vtime is parsed into the vtime type, so a malformed value is rejected
# before any API call.
mock-server 200 '{}'
! snouty debug --input-hash abc --run-id run-1 --vtime not-a-number
stderr 'invalid value .*not-a-number.*'
stderr 'vtime is not a finite decimal number'

# Old-style raw --antithesis.* args are not accepted by `debug`.
! snouty debug --antithesis.debugging.input_hash abc --antithesis.debugging.run_id run --antithesis.debugging.vtime 123
stderr 'unexpected argument.*'

# The target run must be identified by exactly one of --run-id or
#    --session-id. Supplying both, or neither, is an error caught before any
#    API call.
! snouty debug --input-hash abc --vtime 1 --run-id r-1 --session-id s-1
stderr 'specify exactly one of --run-id / --session-id.*'

! snouty debug --input-hash abc --vtime 1
stderr 'specify --run-id or --session-id.*'

# Parameters are provided as typed flags. Before sending, the resolved
#    parameters are printed to stderr with sensitive values redacted. The
#    target run is identified by --run-id (preferred for new integrations).
#    --title sends antithesis.test_name, the session's title.
#    The vtime is normalized to its exact print, so every route into a moment
#    (flags, raw JSON) sends the same text: `1234567890` becomes
#    `1234567890.0`.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890 --title 'stalled leader election' --description 'debug this moment' --recipients team@example.com
stderr '"antithesis.debugging.input_hash": "abc123"'
stderr '"antithesis.debugging.run_id": "9043254f65c9c65d63fe043a0abfc7fc-53-1"'
stderr '"antithesis.debugging.vtime": "1234567890.0"'
stderr '"antithesis.test_name": "stalled leader election"'
stderr '"antithesis.event_description": "debug this moment"'
stderr '"antithesis.report.recipients": "\[REDACTED\]"'
# The mock-server directive configures username/password credentials, which are
# deprecated — the launch warns and steers the user to `snouty login`.
stderr 'warning: username/password authentication is deprecated'

# --param adds custom parameters to the session. Unknown keys, antithesis.*
#    or user-defined, pass through. snouty redacts sensitive values in the
#    printed parameters.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890 --param attrs.team=payments --param antithesis.brand_new_param=on --param antithesis.integrations.github.token=secret
stderr '"attrs.team": "payments"'
stderr '"antithesis.brand_new_param": "on"'
stderr '"antithesis.integrations.github.token": "\[REDACTED\]"'
stdout 'Debugging session started: run_id run-123'

# --param cannot override the value of a dedicated flag.
mock-server 200 '{"message":"Success","runId":"run-123"}'
! snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890 --title mine --param antithesis.test_name=other
stderr "'antithesis.test_name' cannot be overridden via --param"
! stderr 'Requesting the Antithesis multiverse debugger'

# Every parameter value must be a string, custom ones included. A
#    non-string value on stdin fails validation before any API call, with
#    one note for each bad value.
mock-server 200 '{"message":"Success","runId":"run-123"}'
stdin debug_number_param.txt
! snouty debug --stdin
stderr 'validation failed'
stderr 'attrs.retries: 3 is not of type .string.'
stderr -count=1 '^Note: antithesis.debugging.input_hash: 123 is not of type .string.'
! stderr 'Requesting the Antithesis multiverse debugger'

# The live API answers success with HTTP 200. snouty also accepts the
#    documented 202.
mock-server 202 '{"statusCode":202,"runId":"run-123"}'
snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890
stdout 'Debugging session started: run_id run-123'

# Before the launch, the run that --run-id names is looked up. The launch
#    endpoint accepts an unknown run and starts a billable session, so an
#    unknown run gives the shared "run not found" error and nothing is
#    launched. The resolved parameters are printed only just before the
#    launch request, so their absence shows that no launch was sent.
mock-runs-server
! snouty debug --input-hash abc123 --run-id no-such-run --vtime 1234567890
stderr 'run not found: no-such-run'
! stderr 'Requesting the Antithesis multiverse debugger'
! stdout 'Debugging session started'

# The input hash is checked too, with one events search on the run. The
#    launch endpoint accepts a hash that the run never reached and starts a
#    billable session. The hash matches when the run has an event under it.
#    The vtime is not checked: a valid vtime can be rounded, and can sit
#    between the hash's events or after its last. The mock run has events
#    under hash 5181922178177328213 at vtimes 400.5 to 402.0.
mock-runs-server
snouty debug --input-hash 5181922178177328213 --run-id run-1 --vtime 404
stderr 'Requesting the Antithesis multiverse debugger'
stdout 'Debugging session started: run_id mock-run-id'

# An input hash with no event in the run gives an error, and nothing is
#    launched. The hash is compared as text, so a hash that differs only in
#    its last digit does not match.
mock-runs-server
! snouty debug --input-hash 5181922178177328214 --run-id run-1 --vtime 401.6
stderr 'no moment matches input hash 5181922178177328214 in run run-1'
stderr -literal 'Suggestion: search for a moment using `snouty runs events run-1 <search query>`'
! stderr 'Requesting the Antithesis multiverse debugger'
! stdout 'Debugging session started'

# A run that is still starting has no searchable events, so its input hash
#    cannot be checked. snouty refuses the run, and nothing is launched.
mock-runs-server
! snouty debug --input-hash 5181922178177328213 --run-id session-starting --vtime 404
stderr 'run session-starting is still starting; debug it once it is in progress or complete'
stderr -literal 'Suggestion: wait for it with `snouty runs wait session-starting --until running`'
! stderr 'Requesting the Antithesis multiverse debugger'
! stdout 'Debugging session started'

# A run that is in progress has searchable events, so the input hash is
#    checked and the session is launched. run-2 is in progress.
mock-runs-server
snouty debug --input-hash 5181922178177328213 --run-id run-2 --vtime 404
stderr 'Requesting the Antithesis multiverse debugger'
stdout 'Debugging session started: run_id mock-run-id'
mock-runs-server
! snouty debug --input-hash 5181922178177328214 --run-id run-2 --vtime 404
stderr 'no moment matches input hash 5181922178177328214 in run run-2'
env ANTITHESIS_API_KEY=

# The runs endpoints refuse a username and password, so with them the run
#    and the input hash are not looked up. The single-response mock answers
#    every request with the launch response, which a run lookup cannot parse.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty debug --input-hash abc123 --run-id no-such-run --vtime 1234567890
stdout 'Debugging session started: run_id run-123'

# --session-id remains supported for backwards compatibility. The input
#    hash is not looked up for a session.
mock-server 200 '{"message":"Success","runId":"run-123"}'
snouty debug --input-hash abc123 --session-id d5d1c9cf0e64b5dd69b00e97b07fe3f4-18-1 --vtime 1234567890
stderr '"antithesis.debugging.session_id": "d5d1c9cf0e64b5dd69b00e97b07fe3f4-18-1"'

# Parameters can alternatively be read from stdin via --stdin, as JSON or
#    JSON5. A stdin vtime is normalized the same way the flags' is.
mock-server 200 '{"message":"Success","runId":"run-123"}'
stdin debug_json.txt
snouty debug --stdin
stderr '"antithesis.debugging.input_hash": "abc"'
stderr '"antithesis.debugging.vtime": "123.0"'

# A vtime that isn't a number is rejected wherever it came from — here from
# raw JSON on stdin, which no typed flag validated.
mock-server 200 '{}'
stdin debug_bad_vtime.txt
! snouty debug --stdin
stderr 'vtime is not a finite decimal number'
stderr 'antithesis.debugging.vtime = "not-a-number"'

# When both stdin and CLI args are provided, CLI args take priority over
#    stdin values.
mock-server 200 '{"message":"Success","runId":"run-123"}'
stdin debug_run_json.txt
snouty debug --stdin --input-hash override --recipients team@example.com
stderr '"antithesis.debugging.run_id": "9043254f65c9c65d63fe043a0abfc7fc-53-1"'
stderr '"antithesis.debugging.input_hash": "override"'
stderr '"antithesis.debugging.vtime": "329.8037810830865"'
stderr '"antithesis.report.recipients": "\[REDACTED\]"'

# A --param value also takes priority over the same key on stdin.
mock-server 200 '{"message":"Success","runId":"run-123"}'
stdin debug_run_json.txt
snouty debug --stdin --param antithesis.debugging.input_hash=from-param
stderr '"antithesis.debugging.input_hash": "from-param"'

# On success, a confirmation with the run_id is printed to stdout. The
#    session takes a few minutes to start, so the next steps follow: the
#    command that waits until the session runs, and the `runs exec` command
#    at the debugged moment. The hint does not say to try again: when the
#    session is still loading, `runs exec` itself says so.
mock-server 200 '{"message":"Success","runId":"run-123"}'
stdin debug_run_json.txt
snouty debug --stdin
stdout 'Debugging session started: run_id run-123'
stdout -literal 'the session takes a few minutes to start. Wait for it with:'
stdout '^  snouty runs wait run-123 --until running$'
stdout -literal 'then run a script at this moment:'
stdout -literal "  snouty runs exec run-123 6057726200491963783 329.8037810830865 '<script>'"
! stdout '(?i)try again'

# With --json, snouty prints its own envelope: the run id, and nothing else.
# The webhook's body-level statusCode is deliberately not re-exported (issue
# #180) — it carries nothing the exit code doesn't, and here it disagrees with
# the response anyway (the mock answers HTTP 200 saying 202). Use --verbose to
# see the response as it arrived.
mock-server 200 '{"statusCode":202,"runId":"run-123"}'
stdin debug_run_json.txt
snouty --json debug --stdin
stdout '\s*"runId": "run-123".*'
! stdout '.*statusCode.*'
! stdout 'snouty runs'

# Gateway-level rejections (auth, rate limiting) on the launch/debug endpoints
# return an EMPTY body rather than the Launch_Error_Response envelope — verified
# against the live API, which answers a bad token with 401/403 and no payload.
# The HTTP status is the only thing the server said, so it is the whole message.
mock-server 401 ''
! snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890
stderr 'API error: 401 Unauthorized.*'

# An error body snouty cannot make sense of (here an intermediary's HTML page)
# never masks the status either: it is reported with the body shown for
# debugging.
mock-server 502 '<html><body>Bad Gateway</body></html>'
! snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890
stderr 'API error: 502 Bad Gateway.*'
stderr '.*<html><body>Bad Gateway</body></html>.*'

# ANTITHESIS_EXTRA_HEADERS adds arbitrary headers to every API request (for
#    consumption by proxies). With --verbose, the merged request headers are
#    logged to stderr, so a configured extra header shows up on the wire.
mock-server 200 '{"message":"Success","runId":"run-123"}'
set-env ANTITHESIS_EXTRA_HEADERS X-Snouty-Test: hello-world
snouty --verbose debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890
stderr '> x-snouty-test: hello-world.*'

# A malformed ANTITHESIS_EXTRA_HEADERS entry (missing the ':' separator) is
#    rejected when the client is built, before any API call.
mock-server 200 '{"message":"Success","runId":"run-123"}'
set-env ANTITHESIS_EXTRA_HEADERS not-a-valid-header-line
! snouty debug --input-hash abc123 --run-id 9043254f65c9c65d63fe043a0abfc7fc-53-1 --vtime 1234567890
stderr 'ANTITHESIS_EXTRA_HEADERS entry missing \':\' separator: "not-a-valid-header-line".*'
