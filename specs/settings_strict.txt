# Strict settings-file typing
#
# A settings value of the wrong TOML type is a hard error at startup (every
# command resolves settings up front), naming the offending key — rather than
# being silently dropped and resurfacing later as a confusing "not set".

# The error for a non-string value names the key and the file. The
# suggestion names the file.
! snouty --settings integer.toml settings show
stderr '^Error: invalid setting `tenant` in /.*/integer\.toml: must be a string, but found integer$'
stderr 'Suggestion: fix or delete /.*/integer\.toml$'

# A base_url, https_proxy or container_engine that does not parse fails at
# startup, and the error names its source.
env ANTITHESIS_BASE_URL=example.com
! snouty --settings valid.toml settings show
stderr '^Error: invalid ANTITHESIS_BASE_URL: '
stderr -literal 'Suggestion: fix or unset ANTITHESIS_BASE_URL'
env ANTITHESIS_BASE_URL=
env ANTITHESIS_HTTPS_PROXY=http://
! snouty --settings valid.toml settings show
stderr '^Error: invalid ANTITHESIS_HTTPS_PROXY: '
env ANTITHESIS_HTTPS_PROXY=
! snouty --settings engine.toml settings show
stderr '^Error: invalid setting `container_engine` in /.*/engine\.toml: '
stderr 'Suggestion: fix or delete /.*/engine\.toml$'

-- integer.toml --
tenant = 123
-- valid.toml --
tenant = "acme"
-- engine.toml --
container_engine = "none"
