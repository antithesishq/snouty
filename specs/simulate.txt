# Simulation is opt-in and never starts a VM during these argument checks.
env SNOUTY_UNSTABLE_FEATURES=
snouty --help
! stdout 'simulate +Run'
! snouty simulate empty --environment-image environment:latest
stderr 'is an unstable feature and is not enabled'
stderr 'SNOUTY_UNSTABLE_FEATURES=simulate'

env SNOUTY_UNSTABLE_FEATURES=simulate

! snouty simulate empty --environment-image environment:latest --memory 0
stderr '0 is not in 1\.\.=15000'
! snouty simulate empty --environment-image environment:latest --memory 15001
stderr '15001 is not in 1\.\.=15000'
! snouty simulate empty --environment-image environment:latest --disable-node-kills
stderr 'unexpected argument'
! snouty simulate empty --environment-image environment:latest --platform other
stderr 'invalid value'

# Fault settings are parsed before booting the simulation environment.
! snouty simulate empty --node-kills

! snouty simulate empty --fault-tolerance unknown

! snouty simulate empty --notebook-flag unpause
stderr 'unexpected argument'
! snouty simulate empty --custom-scripts '[]'
stderr 'unexpected argument'
! snouty simulate empty --reset-custom-scripts
stderr 'unexpected argument'
! snouty simulate empty --fault-tolerance none --fault-exclusions ''
stderr 'antithesis.fault_exclusions is empty'
! snouty simulate empty --fault-exclusions '('
stderr 'not a valid regular expression'
! snouty simulate --attach simulate-abc --fault-exclusions x
stderr 'cannot be used with'
! snouty simulate --shell --fault-exclusions x
stderr 'cannot be used with'
