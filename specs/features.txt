# Opt-in features
#
# A feature can gate a command that isn't ready to be on for everyone —
# because the Antithesis API it calls is still changing shape, or because most
# tenants can't serve it yet. Gating lets such a command ship in a release
# without putting it in front of users who would only hit a wall.
#
# Features are enabled by id in SNOUTY_UNSTABLE_FEATURES, a comma-separated list. They
# are deliberately an environment variable and not a setting: the gate decides
# which subcommands the parser has, so it must be known before the command
# line is parsed, and a setting can't be read that early without first parsing
# --settings/--profile — which would mean parsing the command line twice.

# 1. An unknown id does not break snouty: every snouty on the machine shares
#    one exported SNOUTY_UNSTABLE_FEATURES, so it can hold an id from a newer
#    build or a retired one.
env SNOUTY_UNSTABLE_FEATURES=from-the-future,runs-exec
! snouty runs exec run-1 -123 1.0 'uname -a'
stderr 'No Antithesis credentials found.*'

# 2. `snouty settings show` omits unknown ids. Both ids listed here are
#    unknown, so the row does not print.
env SNOUTY_UNSTABLE_FEATURES=runs-search,runs-exec
snouty settings show
stdout '^\s*update_channel\s'
! stdout '^\s*features\s'
snouty settings show --json
! stdout '"features"'
! stdout 'runs-search|runs-exec'

# 3. An empty value counts as unset, so the row does not print.
env SNOUTY_UNSTABLE_FEATURES=
snouty settings show
! stdout 'features.*'

# 4. A settings file cannot enable a feature: snouty ignores the `features`
#    key like any unknown key.
snouty --settings features.toml settings show
! stdout '^features'

-- features.toml --
features = ["runs-exec"]
tenant = "from-file"
