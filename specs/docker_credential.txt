# snouty docker-credential: the Docker credential helper protocol for the
# tenant registry, registry.<tenant>.antithesis.com.
#
# Docker writes the server URL on stdin and reads JSON from stdout. The
# username is always snouty-token, and the secret is the Antithesis
# credential of the profile whose registry agrees with the requested host.

# The settings and the credentials in this file are local fixtures.
[staging] skip

env RUST_LOG=

# Docker starts the helper in its own working directory, such as a cloned
# repository. The ./.snouty.toml in this directory sets a hostile tenant,
# repository and base_url, and every case in this file ignores it.

# get: the top-level defaults of the global settings file use
# registry.acme.antithesis.com, and the defaults have their own stored API
# key.
isolate-home stored
substitute ~/.config/snouty/credentials.toml
stdin acme.txt
snouty docker-credential get
stdout -count=1 '^\{"ServerURL":"registry.acme.antithesis.com","Username":"snouty-token","Secret":"${FAKE_API_KEY_A@R}"\}$'
! stderr .

# A scheme, a path and capital letters do not change the host. ServerURL
# echoes the request as Docker gave it.
stdin beta_url.txt
snouty docker-credential get
stdout '^\{"ServerURL":"https://Registry.beta.antithesis.com/v2/","Username":"snouty-token","Secret":"${FAKE_API_KEY_B@R}"\}$'

# A profile whose repository is on another registry does not use its
# tenant registry: the exact "not found" message, exit status 1, and no
# noise on stderr.
stdin gamma.txt
! snouty docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stdout '\{'
! stderr .

# A host that no profile uses is not found either.
stdin elsewhere.txt
! snouty docker-credential get
stdout '^credentials not found in native keychain$'

# The ./.snouty.toml registry is not found: the helper does not read that
# file, so it sends no credential to its base_url.
stdin evil.txt
! snouty docker-credential get
stdout -count=1 '^credentials not found in native keychain$'
! stdout '${FAKE_API_KEY_A@R}'
! stderr .

# The selected profile wins when it agrees with the host.
env ANTITHESIS_PROFILE=beta
stdin beta_url.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'

# When the selected profile does not agree, the defaults win.
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'

# A selected profile that fails to resolve does not stop the search: the
# other profiles still serve get.
env ANTITHESIS_PROFILE=broken
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'
env ANTITHESIS_PROFILE=

# Two named profiles that agree with the host are ambiguous. The error goes
# to stdout, where Docker shows it, and in full to stderr. A settings file
# that SNOUTY_SETTINGS_PATH names is read.
env SNOUTY_SETTINGS_PATH=ambiguous.toml
stdin zeta.txt
! snouty docker-credential get
stdout '^more than one profile uses the registry registry.zeta.antithesis.com: one, two$'
stderr 'set ANTITHESIS_PROFILE'
env SNOUTY_SETTINGS_PATH=

# ANTITHESIS_API_KEY wins over a stored credential, as for every command.
set-env ANTITHESIS_API_KEY ${FAKE_API_KEY_B}
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_B@R}"'
env ANTITHESIS_API_KEY=

# Environment settings apply to every profile, so they serve a registry
# that no settings file names.
env ANTITHESIS_TENANT=delta
env ANTITHESIS_REPOSITORY=registry.delta.antithesis.com/app
stdin delta.txt
snouty docker-credential get
stdout '"ServerURL":"registry.delta.antithesis.com".*"Secret":"${FAKE_API_KEY_A@R}"'
env ANTITHESIS_TENANT=
env ANTITHESIS_REPOSITORY=

# A request with no host fails.
stdin blank.txt
! snouty docker-credential get
stdout '^invalid server URL on stdin: the server URL names no host$'

# store and list fail with one line on stdout, where the engine shows it.
# Docker and Podman call store for a login, and snouty owns the credential.
# Neither engine calls list for a "credHelpers" entry.
substitute store.json
stdin store.json
! snouty docker-credential store
stdout -count=1 '^snouty gives the credential for this registry, so a login is not necessary; to sign in, run `snouty login` or set ANTITHESIS_API_KEY$'
! snouty docker-credential list
stdout -count=1 '^list is not necessary: Docker and Podman do not call it for a "credHelpers" entry$'

# erase reads stdin, erases nothing, and succeeds, because a failed erase
# breaks `podman logout --all` for every registry.
stdin acme.txt
snouty docker-credential erase
! stdout .

# They change nothing: get still serves the stored credential.
stdin acme.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_API_KEY_A@R}"'

# An action outside the protocol fails as a usage error.
! snouty docker-credential fetch
stderr 'invalid value .fetch.'

# OAuth: each get refreshes the token, also when the stored token is still
# valid, and stores the new token.
mock-runs-server
env ANTITHESIS_API_KEY=
env ANTITHESIS_REPOSITORY=registry.testtenant.antithesis.com/app
isolate-home oauth
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_B@R}"'
file ~/.config/snouty/credentials.toml 'antithesis_token = "${FAKE_ACCESS_TOKEN_B@R}"'
file ~/.config/snouty/credentials.toml 'refresh_token = "${FAKE_REFRESH_TOKEN_A@R}"'

# When the refresh fails, the stored token serves while it has not expired,
# with a warning that does not show the token.
env ANTITHESIS_BASE_URL=http://127.0.0.1:9
isolate-home oauth_refresh_fails
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_A@R}"'
stderr '^warning: the OAuth token refresh failed, so the stored token is used until it expires: '
! stderr '${FAKE_ACCESS_TOKEN_A@R}'

# When the stored token has expired too, the refresh error is the result.
isolate-home oauth_expired_refresh
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
! snouty docker-credential get
stdout '^error sending request for url \(http://127.0.0.1:9/auth/cli/refresh\)$'
! stdout '"Secret"'
! stderr 'warning:'
env ANTITHESIS_BASE_URL=

# Without a refresh token, the stored token serves while it has not expired.
isolate-home oauth_no_refresh
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
snouty docker-credential get
stdout '"Secret":"${FAKE_ACCESS_TOKEN_A@R}"'

# An expired token without a refresh token fails, and names the fix.
isolate-home oauth_expired
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
! snouty docker-credential get
stdout '^the stored OAuth token has expired, and there is no refresh token to replace it$'
stderr 'run `snouty login`'

# The registry takes an Antithesis token, so a username and password fail.
isolate-home password
substitute ~/.config/snouty/credentials.toml
stdin testtenant.txt
! snouty docker-credential get
stdout 'does not accept username/password'

# So does a GitHub Actions OIDC token, which is GitHub's token.
isolate-home oidc
env ACTIONS_ID_TOKEN_REQUEST_URL=http://127.0.0.1:9/token?api-version=2.0
set-env ACTIONS_ID_TOKEN_REQUEST_TOKEN ${FAKE_OIDC_REQUEST_TOKEN}
stdin testtenant.txt
! snouty docker-credential get
stdout '^the Antithesis container registry does not accept a GitHub Actions OIDC token$'
stderr 'set ANTITHESIS_API_KEY'
