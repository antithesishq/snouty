# snouty launch --config
#
# The --config/-c flag points to a local directory containing a
# docker-compose.yaml file that will be pushed as a config image.

# Without a registry setting, any error of the registry endpoint names the fix.
[!staging] set-env ANTITHESIS_API_KEY ${FAKE_API_KEY_A}
[!staging] mock-server 403 '{"error":"Forbidden"}'
[!staging] ! snouty launch --launcher basic_test -c config
[!staging] stderr 'no container registry is configured'
[!staging] stderr -literal 'Suggestion: run `snouty login`, or set the registry setting (see https://github.com/antithesishq/snouty#settings)'
[!staging] mock-server 404 '{"error":"Not Found"}'
[!staging] ! snouty launch --launcher basic_k8s_test -c k8s_config
[!staging] stderr 'no container registry is configured'

# Bad input fails before the request for the registry.
! snouty launch --launcher basic_test -c empty
stderr '.*does not contain.*docker-compose.yaml.*manifests/.*'

# Rejects a registry setting that is not a registry before it builds anything.
env 'ANTITHESIS_REGISTRY=not a registry'
! snouty launch --launcher basic_test -c config
stderr -literal 'invalid ANTITHESIS_REGISTRY: `not a registry` is not a container registry'
stderr -literal 'Suggestion: fix or unset ANTITHESIS_REGISTRY'

env ANTITHESIS_REGISTRY=registry.example.com/repo

# Rejects a nonexistent directory.
! snouty launch --launcher basic_test -c /nonexistent/path
stderr 'not a directory.*'

# 2b. Long flag --config also accepted.
! snouty launch --launcher basic_test --config /nonexistent/path
stderr 'not a directory.*'

# Rejects a directory without docker-compose.yaml or manifests/.
! snouty launch --launcher basic_test -c empty
stderr '.*does not contain.*docker-compose.yaml.*manifests/.*'

# Rejects docker-compose.yml (wrong extension).
! snouty launch --launcher basic_test -c yml_dir
stderr '.*rename it to docker-compose.yaml.*'

# Rejects an ambiguous directory containing both compose and manifests/.
! snouty launch --launcher basic_test -c ambiguous
stderr '.*both docker-compose.yaml and a manifests/ subdirectory.*'

# --config conflicts with --config-image (clap).
! snouty launch --launcher basic_test -c config --config-image some-image:latest
stderr '--config.*'

# --allow-remote-images only applies to a local config dir, so it requires
# --config and conflicts with --config-image.
! snouty launch --launcher basic_test --allow-remote-images
stderr 'the following required arguments were not provided'
stderr '--config <CONFIG>'
! snouty launch --launcher basic_test --allow-remote-images --config-image some-image:latest
stderr '--allow-remote-images.* cannot be used with .--config-image'

# --config conflicts with --param antithesis.config_image.
! snouty launch --launcher basic_test -c config --param antithesis.config_image=some-image:latest
stderr 'cannot be overridden via --param.*'
