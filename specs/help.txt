# snouty help and version

# version subcommand prints version string.
snouty version
stdout 'snouty \d+\.\d+\.\d+'

# --version flag prints version string.
snouty --version
stdout 'snouty \d+\.\d+\.\d+'

# --help shows primary subcommands.
snouty --help
stdout '.*debug.*'
stdout '.*version.*'
stdout '.*runs.*'
stdout '.*docs.*'
stdout '.*completions.*'
stdout '.*update.*'
stdout '.*validate.*'
stdout '.*doctor.*'

# Flag descriptions wrap at word boundaries instead of running off the
# terminal, which split them mid-word. Captured output gets the same measure
# as a terminal, so no line here passes 100 columns.
snouty launch --help
! stdout '^.{101,}$'
snouty runs logs --help
! stdout '^.{101,}$'

# validate --help documents both the compose and Kubernetes flows.
snouty validate --help
stdout '.*Compose configs:.*'
stdout '.*docker-compose locally.*'
stdout '.*setup-complete.*'
stdout '.*Test commands are not executed.*'
stdout '.*Kubernetes configs:.*'
stdout '.*k8s-validator.*'
stdout '.*manifests\/.*'
